-- The pinned library's title editor treats its title container as a TextLabel.
local function fixLibraryTitle(source)
    local broken = 'SettingsPage.Window.Items["ActualTitle"].Instance.Text = Value'
    local first, last = source:find(broken, 1, true)
    if first then
        local replacement = [[local container = SettingsPage.Window.Items["ActualTitle"].Instance
                        if container:IsA("TextLabel") then container.Text = Value else
                            local assigned = false
                            for _, child in ipairs(container:GetChildren()) do
                                if child:IsA("TextLabel") then
                                    child.Text = assigned and "" or Value
                                    assigned = true
                                end
                            end
                        end]]
        source = source:sub(1, first - 1) .. replacement .. source:sub(last + 1)
    end
    return source
end

local function syncListener()
    if liveSyncing or liveDesired==liveRegistered then return end
    if type(requestFn)~="function" then liveStatus="executor HTTP unavailable"; return end
    liveSyncing=true
    task.spawn(function()
        while liveDesired~=liveRegistered do
            local target=liveDesired
            local ok,response=pcall(function()
                return requestFn({Url="http://127.0.0.1:8787/live-lyrics",Method="POST",
                    Headers={["Content-Type"]="application/json"},Body=HttpService:JSONEncode({client=liveOwner,enabled=target})})
            end)
            local decoded,result=false,nil
            if ok and type(response)=="table" then decoded,result=pcall(function() return HttpService:JSONDecode(response.Body or response.body) end) end
            local code=ok and type(response)=="table" and (response.StatusCode or response.status_code)
            if not decoded or type(result)~="table" or not result.ok or (code and code~=200) then
                liveStatus=target and "live bridge unavailable" or "off"
                liveRetryAt=os.clock()+5
                break
            end
            liveRegistered=target
            liveStatus=target and "starting" or "off"
        end
        liveSyncing=false
    end)
end
local function setLiveListening(enabled)
    liveDesired=enabled and alive or false
    if not liveDesired then liveData=nil; liveTrackKey=""; liveStatus="off" end
    syncListener()
end

local function parkEnvironment(object)
    if object ~= worldSky and object ~= worldAir and (object:IsA("Sky") or object:IsA("Atmosphere")) then
        savedEnvironment[object] = true
        object.Parent = nil
    end
end
local function restoreWorld()
    worldSky.Parent, worldAir.Parent = nil, nil
    for object in pairs(savedEnvironment) do
        pcall(function() if object.Parent == nil then object.Parent = Lighting end end)
    end
    savedEnvironment = {}
    if savedClock ~= nil and Lighting.ClockTime == imposedClock then Lighting.ClockTime = savedClock end
    savedClock, imposedClock = nil, nil
end
local function applyWorld()
    if not settings.world then restoreWorld(); return end
    if savedClock == nil then savedClock = Lighting.ClockTime end
    for _, object in ipairs(Lighting:GetChildren()) do parkEnvironment(object) end
    local preset = worldPresets[settings.worldPreset] or worldPresets["Violet night"]
    worldSky.StarCount, worldSky.CelestialBodiesShown = settings.stars, true
    worldAir.Color, worldAir.Decay = preset.color, preset.decay
    worldAir.Density, worldAir.Haze, worldAir.Glare = settings.haze, 1.4, 0.15
    imposedClock = preset.time
    Lighting.ClockTime = imposedClock
    worldSky.Parent, worldAir.Parent = Lighting, Lighting
end

-- A local cosmetic rig follows the real character, preserving movement and camera behavior.
local morph = {ticket = 0, model = nil, root = nil, anchor = nil, hidden = {}, parts = {}, connections = {}, pose = nil, poseDirty = false, unsafe = false, vertical = 0, direction = Vector3.new(0,0,-1)}
local MorphStatus, CollisionStatus
local function morphStatus(message) if alive and MorphStatus then MorphStatus:SetText(message) end end
local function clearMorph()
    morph.ticket = morph.ticket + 1
    for _, connection in ipairs(morph.connections) do connection:Disconnect() end
    morph.connections = {}
    if morph.model then morph.model:Destroy() end
    if morph.anchor then morph.anchor:Destroy() end
    morph.model, morph.root, morph.anchor, morph.pose, morph.poseDirty, morph.unsafe, morph.vertical = nil, nil, nil, nil, false, false, 0
    morph.parts = {}
    for object, saved in pairs(morph.hidden) do
        pcall(function() if object[saved.property] == 1 then object[saved.property] = saved.value end end)
    end
    morph.hidden = {}
    if CollisionStatus and alive then CollisionStatus:SetText("Preset collisions: inactive") end
end
local function checkMorphCollisions()
    if not CollisionStatus or not alive then return end
    if not morph.model then CollisionStatus:SetText("Preset collisions: inactive"); return end
    local total,solid,loose=0,0,0
    local ok=pcall(function()
        for _,part in ipairs(morph.model:GetDescendants()) do
            if part:IsA("BasePart") then
                total=total+1
                if part.CanCollide then solid=solid+1 end
                if not part.Anchored then loose=loose+1 end
            end
        end
    end)
    CollisionStatus:SetText(ok and string.format("Parts %d | solid %d | loose %d",total,solid,loose) or "Cannot inspect preset parts")
end
-- Enforce the ghost state before physics and whenever a new cosmetic part appears.
local function setGhostPart(part)
    local ok,valid=pcall(function()
        if part.CanCollide then part.CanCollide=false end
        if part.CanTouch then part.CanTouch=false end
        if part.CanQuery then part.CanQuery=false end
        if not part.Anchored then part.Anchored=true end
        if not part.Massless then part.Massless=true end
        return not part.CanCollide and not part.CanTouch and not part.CanQuery and part.Anchored and part.Massless
    end)
    return ok and valid
end
local function stopUnsafeMorph()
    clearMorph()
    morphStatus("Morph stopped: collision settings unavailable")
end
local function enforceMorphParts()
    local model=morph.model
    if not model then return true end
    if not model.Parent then clearMorph(); morphStatus("Morph removed"); return false end
    for part in pairs(morph.parts) do
        local ok,belongs=pcall(function() return part:IsDescendantOf(model) end)
        if not ok or not belongs then morph.parts[part]=nil
        elseif not setGhostPart(part) then stopUnsafeMorph(); return false end
    end
    return true
end

    local ok,signal=pcall(function() return RunService[name] end)
    if not ok or not signal then return false end
    return pcall(function() connect(signal,function() if alive then enforceMorphParts() end end) end)
end
local physicsGuardAvailable=installPhysicsGuard("PreSimulation") or installPhysicsGuard("Stepped")

-- Keep the cosmetic rig upright even when the source root leans or spins.
local function morphFrame(root)
    local anchor = morph.anchor
    local transform = anchor and anchor.WorldCFrame or root.CFrame
    local position = (anchor and anchor.WorldPosition or root.Position) + Vector3.new(0,morph.vertical+settings.morphOffset,0)
    local look = transform.LookVector
    local flat = Vector3.new(look.X, 0, look.Z)
    if flat.Magnitude > 0.001 then morph.direction = flat.Unit end
    return CFrame.lookAt(position, position + morph.direction)
end
-- Fully anchored cosmetic parts are posed explicitly; no cloned assembly can push the player.
local function groundDistance(model, root)
    local humanoid=model:FindFirstChildOfClass("Humanoid")
    if humanoid and type(humanoid.HipHeight)=="number" and root.Size then
        local distance=root.Size.Y/2+humanoid.HipHeight
        if humanoid.RigType==Enum.HumanoidRigType.R6 then
            local leg=model:FindFirstChild("Left Leg") or model:FindFirstChild("Right Leg")
            if leg and leg:IsA("BasePart") then distance=distance+leg.Size.Y end
        end
        if distance>0 and distance<25 then return distance end
    end
    local bottom
    for _,name in ipairs({"LeftFoot","RightFoot","Left Leg","Right Leg"}) do
        local foot=model:FindFirstChild(name)
        if foot and foot:IsA("BasePart") then
            local y=foot.Position.Y-foot.Size.Y/2
            bottom=bottom and math.min(bottom,y) or y
        end
    end
    return bottom and (root.Position.Y-bottom) or nil
end
local function buildMorphPose(model,cloneRoot,character)
    local members,parts,edges,sources = {},{},{},{}
    for _,object in ipairs(character:GetDescendants()) do
        if object:IsA("Motor6D") and object.Parent then sources[object.Parent.Name.."/"..object.Name]=object end
    end
    for _,object in ipairs(model:GetDescendants()) do
        if object:IsA("BasePart") then members[object]=true; table.insert(parts,object) end
    end
    for _,joint in ipairs(model:GetDescendants()) do
        local a,b,kind
        if joint:IsA("Motor6D") or joint:IsA("Weld") or joint:IsA("ManualWeld") then
            a,b,kind=joint.Part0,joint.Part1,joint:IsA("Motor6D") and "motor" or "weld"
        elseif joint:IsA("WeldConstraint") then a,b,kind=joint.Part0,joint.Part1,"rigid"
        elseif joint:IsA("RigidConstraint") and joint.Attachment0 and joint.Attachment1 then
            a,b,kind=joint.Attachment0.Parent,joint.Attachment1.Parent,"constraint"
        end
        if a and b and members[a] and members[b] and a~=b then
            local edge={joint=joint,a=a,b=b,kind=kind}
            if kind=="rigid" then edge.offset=a.CFrame:ToObjectSpace(b.CFrame)
            elseif kind=="constraint" then edge.offset=joint.Attachment0.CFrame*joint.Attachment1.CFrame:Inverse()
            elseif kind=="motor" and joint.Name~="Root" and joint.Name~="RootJoint" and joint.Parent then
                edge.source=sources[joint.Parent.Name.."/"..joint.Name]
            end
            table.insert(edges,edge)
        end
    end
    local known,ordered={[cloneRoot]=true},{}
    local changed=true
    while changed do
        changed=false
        for _,edge in ipairs(edges) do
            if not edge.used and known[edge.a] ~= known[edge.b] then
                edge.forward=not not known[edge.a]; edge.used=true
                known[edge.forward and edge.b or edge.a]=true
                table.insert(ordered,edge); changed=true
            end
        end
    end
    local detached={}
    for _,part in ipairs(parts) do
        if not known[part] then table.insert(detached,{part=part,offset=cloneRoot.CFrame:ToObjectSpace(part.CFrame)}) end
    end
    return {root=cloneRoot,edges=ordered,detached=detached}
end
local function applyMorphPose(pose, target)
    pose.root.CFrame=target
    for _,edge in ipairs(pose.edges) do
        if edge.a.Parent and edge.b.Parent and edge.joint.Parent then
            local relative=edge.offset
            if not relative then
                local transform=CFrame.new()
                if edge.kind=="motor" and edge.source and edge.source.Parent then transform=edge.source.Transform end
                relative=edge.joint.C0*transform*edge.joint.C1:Inverse()
            end
            if edge.forward then edge.b.CFrame=edge.a.CFrame*relative else edge.a.CFrame=edge.b.CFrame*relative:Inverse() end
        end
    end
    for _,part in ipairs(pose.detached) do
        if part.part.Parent then part.part.CFrame=target*part.offset end
    end
end
local function loadMorph(outfitId)
    clearMorph()
    if not physicsGuardAvailable then morphStatus("Morph unavailable: physics guard missing"); return end
    local ticket, target = morph.ticket, settings.morphTarget:match("^%s*(.-)%s*$")
    if not outfitId and target == "" then morphStatus("Enter a username or user ID"); return end
    morphStatus("Loading avatar...")
    task.spawn(function()
        local model, description
        local success, failure = pcall(function()
            local player = Players.LocalPlayer
            local character = player and player.Character
            local humanoid = character and character:FindFirstChildOfClass("Humanoid")
            local root = character and character:FindFirstChild("HumanoidRootPart")
            assert(humanoid and root, "Character unavailable")
            if outfitId then
                description = Players:GetHumanoidDescriptionFromOutfitId(outfitId)
            else
                local userId = tonumber(target) or Players:GetUserIdFromNameAsync(target)
                description = Players:GetHumanoidDescriptionFromUserIdAsync(userId)
            end
            if not alive or morph.ticket ~= ticket then return end
            -- Catalog body bundles are R15; custom avatars use the player's own rig.
            model = Players:CreateHumanoidModelFromDescriptionAsync(description, outfitId and Enum.HumanoidRigType.R15 or humanoid.RigType)
            if not alive or morph.ticket ~= ticket or player.Character ~= character then return end
            local cloneHumanoid = model:FindFirstChildOfClass("Humanoid")
            local cloneRoot = model:FindFirstChild("HumanoidRootPart")
            assert(cloneHumanoid and cloneRoot, "Avatar unavailable")
            model.Name, model.PrimaryPart = "LarpsenseAvatar", cloneRoot
            local ownedModel = model
            local function prepare(object)
                if object:IsA("BasePart") then
                    morph.parts[object]=true
                    if not setGhostPart(object) then error("cosmetic collision settings unavailable") end
                elseif object:IsA("Script") or object:IsA("LocalScript") then object:Destroy() end
            end
            for _, object in ipairs(model:GetDescendants()) do prepare(object) end
            table.insert(morph.connections, ownedModel.DescendantAdded:Connect(function(object)
                local safe=pcall(prepare,object)
                if not safe then
                    if morph.model==ownedModel then stopUnsafeMorph() else morph.unsafe=true end
                    return
                end
                if morph.model == ownedModel then morph.poseDirty=true end
            end))
            if ownedModel.DescendantRemoving then
                table.insert(morph.connections, ownedModel.DescendantRemoving:Connect(function(object)
                    morph.parts[object]=nil
                    if morph.model==ownedModel then morph.poseDirty=true end
                end))
            end
            cloneHumanoid.DisplayDistanceType = Enum.HumanoidDisplayDistanceType.None
            cloneHumanoid.AutoRotate = false
            -- Only the cosmetic humanoid is disabled; the real character is untouched.
            pcall(function() cloneHumanoid.EvaluateStateMachine = false end)
            pcall(function() cloneHumanoid:ChangeState(Enum.HumanoidStateType.Physics) end)
            -- A local attachment tracks the real root without joining its physics assembly.
            local anchor = Instance.new("Attachment")
            anchor.Name, anchor.Parent = "LarpsenseMorphAnchor", root
            morph.anchor = anchor
            local realGround,cloneGround=groundDistance(character,root),groundDistance(model,cloneRoot)
            morph.vertical=realGround and cloneGround and (cloneGround-realGround) or 0
            model:PivotTo(morphFrame(root))
            morph.pose=buildMorphPose(model,cloneRoot,character)
            model.Parent = workspace
            morph.model, morph.root = model, root
            model = nil
            assert(not morph.unsafe, "cosmetic collision settings unavailable")
            local function hide(object)
                if morph.hidden[object] then return end
                local property
                if object:IsA("BasePart") then property = "LocalTransparencyModifier"
                elseif object:IsA("Decal") or object:IsA("Texture") then property = "Transparency" end
                if property then
                    morph.hidden[object] = {property=property, value=object[property]}
                    object[property] = 1
                end
            end
            for _, object in ipairs(character:GetDescendants()) do hide(object) end
            table.insert(morph.connections, character.DescendantAdded:Connect(hide))
            morphStatus("Avatar applied (local)")
            checkMorphCollisions()
        end)
        if description then description:Destroy() end
        if model then model:Destroy() end
        if not success and alive and morph.ticket == ticket then
            clearMorph()
            if tostring(failure):find("cosmetic collision settings unavailable",1,true) then
                morphStatus("Morph stopped: collision settings unavailable")
            else morphStatus("Avatar unavailable — outfit or rig could not load") end
        end
    end)
end

    for _, name in ipairs({"highlight", "trail", "a", "b", "wind", "aura", "windAttachment"}) do

    if characterFx.character ~= character or characterFx.root ~= root then
        clearCharacterFx()

    end

        local highlight = Instance.new("Highlight")

        characterFx.highlight.Enabled, characterFx.highlight.Adornee = settings.outline, morph.model or character
        characterFx.highlight.OutlineColor = accent

        aura.Size, aura.Transparency = NumberSequence.new(0.18,0), NumberSequence.new(0.2,1)
        aura.Parent, characterFx.aura = attachment, aura
    end
    if characterFx.wind then
        characterFx.wind.Enabled, characterFx.wind.Rate, characterFx.wind.Color = settings.wind, settings.windRate, ColorSequence.new(accent)
        characterFx.aura.Enabled, characterFx.aura.Rate, characterFx.aura.Color = settings.aura, settings.windRate/2, ColorSequence.new(accent)
    end

-- Universal player visuals: screen-space elements use Roblox GUI objects, not Drawing API.
local function installUniversalEsp()
    local options = {enabled=false, box=true, names=true, distance=true, health=true,
        highlight=true, skeleton=false, tracer=false, teammates=false, rainbow=false, ping=false,
        pulse=true, teamColor=false, healthNumber=false, healthGradient=true, nameMode="Display name", textSize=13,
        font="Code", textOutline=true, textGap=2, tool=false, teamLabel=false, boxFill=false, tracerOrigin="Bottom",
        boxThickness=1, lineThickness=1, highlightFill=0.45, throughWalls=true, healthFormat="Percent",
        maxDistance=2000, updateRate=20, followSpeed=14, opacity=0.7, color=Color3.fromRGB(255,255,255)}
    local chams = {standalone=false, fill=true, outline=true, colorMode="ESP colour", outlineColor=Color3.fromRGB(255,255,255),
        linkOpacity=true, opacity=0.7, outlineStrength=1, pulseFill=true, pulseOutline=true, pulseSpeed=3, pulseDepth=0.12,
        rainbowSpeed=0.12, rainbowOutlineOffset=false, distanceFade=false, fadeFloor=0.35,
        useEspFilters=true, teammates=false, maxDistance=2000}
    local me = Players.LocalPlayer
    local gui = Instance.new("ScreenGui")
    gui.Name, gui.IgnoreGuiInset, gui.ResetOnSpawn, gui.DisplayOrder = "LarpsenseESP", true, false, 20
    gui.Parent = parent
    -- Snap slots are shared by the preview and the actual ESP.
    local defaults = {name="top-2", distance="bottom-2", health="left", ping="right-2", tool="bottom-2", team="top-2"}
    local layout = {name="top-2", distance="bottom-2", health="left", ping="right-2", tool="bottom-2", team="top-2"}
    local colors = {box=options.color, name=options.color, distance=options.color, ping=options.color,
        detail=options.color, highlight=options.color, tool=options.color, team=options.color, health=Color3.fromRGB(60,205,95)}
    local textOrder = {"name", "distance", "ping", "tool", "team"}
    local function textEnabled(key)
        return (key == "name" and options.names) or (key == "distance" and options.distance) or (key == "ping" and options.ping) or (key == "tool" and options.tool) or (key == "team" and options.teamLabel)
    end
    local function colorFor(key, plr)
        if options.rainbow then return Color3.fromHSV((os.clock()*0.12)%1,0.65,1) end
        if options.teamColor and plr and plr.Team and plr.TeamColor then return plr.TeamColor.Color end
        return colors[key]
    end
    local function chamsColors(plr,hum)
        if chams.colorMode=="ESP colour" then
            local color=colorFor("highlight",plr); return color,color
        elseif chams.colorMode=="Custom" then return colors.highlight,chams.outlineColor
        elseif chams.colorMode=="Team" then
            local team=plr and plr.Team and plr.TeamColor and plr.TeamColor.Color
            return team or colors.highlight,team or chams.outlineColor
        elseif chams.colorMode=="Health" then
            local hp=hum and math.clamp(hum.Health/math.max(hum.MaxHealth or 100,1),0,1) or 1
            local color=Color3.fromRGB(math.floor(255*(1-hp)),math.floor(220*hp),95)
            return color,color
        elseif chams.colorMode=="Rainbow" then
            local hue=(os.clock()*chams.rainbowSpeed)%1
            return Color3.fromHSV(hue,0.65,1),Color3.fromHSV((hue+(chams.rainbowOutlineOffset and 0.15 or 0))%1,0.65,1)
        end
        return colors.highlight,chams.outlineColor
    end
    local function chamsOpacity(distance,limit)
        local base=chams.linkOpacity and options.opacity or chams.opacity
        local fade=chams.distanceFade and (1-math.clamp((distance or 0)/math.max(limit or options.maxDistance,1),0,1)*(1-chams.fadeFloor)) or 1
        return base*fade
    end
    local function chamsPulse()
        return options.pulse and (0.5+0.5*math.sin(os.clock()*chams.pulseSpeed)) or 0
    end
    local function chamsTransparency(distance,limit)
        local alpha,pulse=chamsOpacity(distance,limit),chamsPulse()
        local fill=chams.fill and math.clamp(1-alpha*options.highlightFill+(chams.pulseFill and pulse*chams.pulseDepth or 0),0,1) or 1
        local stroke=chams.outline and math.clamp(1-alpha*chams.outlineStrength+(chams.pulseOutline and pulse*chams.pulseDepth*1.25 or 0),0,1) or 1
        return fill,stroke
    end
    local function textPosition(key,x,y,w,h,textWidth)
        local side,slot = layout[key]:match("^(%a+)%-(%d)$")

    -- Some games replicate their own ping in player attributes or leaderstats.
    -- Values in those fields are conventionally milliseconds unless explicitly suffixed with "s".
    local function gamePingValue(value)
        local n
        if type(value)=="number" then n=value
        elseif type(value)=="string" then
            local cleaned=value:lower():match("^%s*(.-)%s*$")
            local amount,unit=cleaned:match("^(%d+%.?%d*)%s*(%a*)$")
            if not amount or (unit~="" and unit~="ms" and unit~="s") then return nil end
            n=tonumber(amount); if unit=="s" then n=n*1000 end
        end
        if type(n)=="number" and n==n and n>=0 and n<=60000 then return math.floor(n+0.5) end
    end
    local function readPlayerPing(plr)
        local ok,value=pcall(function() return plr:GetNetworkPing() end)
        if ok and type(value)=="number" and value==value and value>=0 and value<60 then return math.floor(value*1000+0.5),"network" end
        for _,name in ipairs({"PingMs","Ping","ping"}) do
            local success,reported=pcall(function() return plr:GetAttribute(name) end)
            local parsed=success and gamePingValue(reported)
            if parsed then return parsed,"game" end
        end
        local function fromContainer(container)
            local success,children=pcall(function() return container:GetChildren() end)
            if not success then return nil end
            for _,item in ipairs(children) do
                local key=item.Name:lower()
                if (key=="ping" or key=="pingms") and (item:IsA("IntValue") or item:IsA("NumberValue") or item:IsA("StringValue")) then
                    local parsed=gamePingValue(item.Value)
                    if parsed then return parsed end
                end

local function sendControl(action)
    if controlPending then return end
    if type(requestFn) ~= "function" or not data or not data.controls or not data.controls[action] then
        controlMessage, controlMessageUntil = "Control unavailable", os.clock() + 3
        return
    end
    controlPending = true
    task.spawn(function()
        local success, response = pcall(requestFn, {Url = "http://127.0.0.1:8787/control", Method = "POST",
            Headers = {["Content-Type"] = "application/json"}, Body = HttpService:JSONEncode({action = action})})
        if not alive then return end
        local decoded, result = false, nil
        if success and type(response) == "table" then
            decoded, result = pcall(function() return HttpService:JSONDecode(response.Body or response.body) end)
        end
        controlMessage = decoded and result and (result.ok and "Command sent" or result.error) or "Control unavailable"
        controlMessageUntil, controlPending = os.clock() + 3, false
    end)
end

local function clock(ms)
    local seconds = math.floor(math.max(0, tonumber(ms) or 0) / 1000)
    return string.format("%d:%02d", math.floor(seconds / 60), seconds % 60)
end
local function lyricAt(lines, seconds)
    local low, high, found = 1, #lines, 0
    while low <= high do
        local mid = math.floor((low + high) / 2)
        if (tonumber(lines[mid].time) or 0) <= seconds then found, low = mid, mid + 1 else high = mid - 1 end
    end
    return found
end
-- Tagged words are absolute seconds; otherwise interpolate evenly within the current line.
local function joinWords(words, last)
    local result=""
    for i=1,last do
        local word=words[i]
        if word and word~="" then
            result=result .. ((result~="" and not word:match("^[%.,!?;:]")) and " " or "") .. word
        end
    end
    return result
end
local function lyricWord(lines,index,seconds,duration,buildLine)
    local line=lines[index]
    if not line then return "","unavailable" end
    local start=tonumber(line.time)
    local following=lines[index+1] and tonumber(lines[index+1].time) or duration
    if not start or not following or following<=start then return "","unavailable" end
    local tagged=line.words
    if type(tagged)=="table" and #tagged>0 then
        local valid,previous,selected,values=true,start,0,{}
        for _,entry in ipairs(tagged) do
            local t=type(entry)=="table" and tonumber(entry.time) or nil
            local value=type(entry)=="table" and entry.text or nil
            if not t or t<previous or t>=following or type(value)~="string" or value=="" then valid=false; break end
            previous=t
            values[#values+1]=value
            if t<=seconds then selected=#values end
        end
        if valid then return buildLine and joinWords(values,selected) or (values[selected] or ""),"tagged" end
    end
    local words={}
    for word in tostring(line.text or ""):gmatch("%S+") do words[#words+1]=word end
    if #words==0 then return "","unavailable" end
    local fraction=math.clamp((seconds-start)/(following-start),0,1)
    local selected=math.min(#words,math.floor(fraction*#words)+1)
    return buildLine and joinWords(words,selected) or words[selected],"estimated"
end

    local playing = data and data.token_ok and data.source == "smtc" and data.playing
    if data and not data.token_ok then
        song, state, line = "Nothing playing", "Idle", "Waiting for a track"
    elseif data and data.token_ok then
        song, artist, album = data.song or "Unknown track", data.artist or "Unknown artist", data.album or ""
        if album == "" then album = "—" end
        state = data.source == "window" and "Detected (no timeline)" or (data.playing and "Playing" or "Paused")
        local length = tonumber(data.duration_ms) or 0
        if data.source == "smtc" and length > 0 then
            local position = tonumber(data.progress_ms) or 0
            if playing then position = position + math.max(0, now - receivedAt) * 1000 end
            position = math.clamp(position, 0, length)
            progress, fraction = clock(position) .. " / " .. clock(length), position / length
            if type(data.lyrics) == "table" and #data.lyrics > 0 then
                local index = lyricAt(data.lyrics, position / 1000 + settings.offset)
                lyricKey = song .. "\0" .. artist .. "\0" .. tostring(index)
                actualLine = index > 0 and (data.lyrics[index].text or "") or ""
                line = actualLine ~= "" and actualLine or "Instrumental"
                following = data.lyrics[index + 1] and data.lyrics[index + 1].text or ""
                shownLyric=actualLine
                if settings.lyricMode~="Line" and index>0 and actualLine~="" then
                    local word
                    word,timing=lyricWord(data.lyrics,index,position/1000+settings.offset,length/1000,settings.lyricMode=="Word line")
                    shownLyric=timing=="unavailable" and actualLine or ((timing=="estimated" and "~" or "") .. (word~="" and word or "…"))
                end
            else
                line = data.lyrics_status == "plain" and "Plain lyrics in menu (no timestamps)" or "Synced lyrics unavailable"
            end
        else
            line = "Timeline unavailable"
        end
    end
    if settings.liveListen then
        local valid = liveData and now-liveReceivedAt<2.5 and currentTrackKey()~="" and liveTrackKey==currentTrackKey()
        local status = valid and tostring(liveData.status or "listening") or liveStatus
        timing, following, actualLine, shownLyric = "live: "..status.." (delayed)", "", "", ""
        line = "Live: "..status
        if valid and type(liveData.words)=="table" then
            local words, selected = {}, 0
            local elapsed = (tonumber(liveData.age_ms) or 0)/1000 + math.max(0,now-liveReceivedAt)
            for _,entry in ipairs(liveData.words) do
                local t=type(entry)=="table" and tonumber(entry.time) or nil
                local value=type(entry)=="table" and entry.text or nil
                if t and t>=0 and type(value)=="string" and value~="" then
                    words[#words+1]=value
                    if t<=elapsed then selected=#words end
                end
            end
            if #words>0 then
                actualLine=joinWords(words,#words)
                if settings.lyricMode=="Line" then shownLyric=actualLine
                elseif settings.lyricMode=="Word" then shownLyric=words[selected] or "…"
                else shownLyric=selected>0 and joinWords(words,selected) or "…" end
                line=shownLyric
                lyricKey="live\0"..tostring(liveData.sequence or 0)
            end
        end
    end
    text(Song, "Track: " .. song); text(Artist, "Artist: " .. artist); text(Album, "Album: " .. album)
    text(Status, "Status: " .. state .. (settings.waves and not spectrumAvailable and " • audio unavailable" or "")); text(Progress, "Progress: " .. progress)
    text(Current, not settings.liveListen and data and data.lyrics_status == "plain" and data.lyrics_plain or line); text(Next, following ~= "" and ("Next: " .. following) or "")
    text(LyricTiming, "Word timing: " .. timing)
    title.Text, artistLabel.Text, timeLabel.Text = song, artist, progress
    playButton.Text = playing and "PAUSE" or "PLAY"
    local caps = data and data.controls or {}
    playButton.TextColor3 = caps.play_pause and textColor or dim
    nextButton.TextColor3 = caps.next and textColor or dim
    progressFill.Size = UDim2.fromScale(fraction, 1)
    local hudMessage = now < controlMessageUntil and controlMessage or
        (settings.lyrics and actualLine ~= "" and ((settings.liveListen and "LIVE " or "")..shownLyric) or
        (settings.liveListen and line or (not data and offline or (spectrumAvailable and state or state .. " • audio unavailable"))))
    if hudStatus.Text~=hudMessage then
        hudStatus.Text=hudMessage
        if settings.lyricMode=="Word line" and actualLine~="" then hudFade=0.35 end
    end
    local newLine = settings.sky and settings.lyrics and shownLyric or ""
    if newLine ~= "" and skyAnchorKey ~= lyricKey then
        skyAnchorKey, skyOrigin = lyricKey, nil
    end
    if skyLine ~= newLine then
        previousSkyLine, skyLine, skyTransition = skyLine, newLine, now
        skyOld.Text, skyNew.Text = previousSkyLine, skyLine
    end
end
local lastTextUpdate = 0
connect(RunService.RenderStepped, function(dt)
    local now = os.clock()
    if now - lastTextUpdate >= 0.1 then lastTextUpdate = now; updateDisplay(now) end
    hudStatus.TextTransparency=hudFade
    hudFade=math.max(0,hudFade-math.min(dt,0.25)*2)
    local accent = Library.Theme["Accent"]
    accentLine.BackgroundColor3, progressFill.BackgroundColor3 = accent, accent
    local active = data and data.token_ok and (data.source == "window" or (data.source == "smtc" and data.playing)) and spectrumAvailable and settings.waves
    for i, bar in ipairs(bars) do
        local target = active and math.clamp(((tonumber(spectrum[i]) or 0) * settings.gain) ^ 0.8, 0, 1) or 0
        local rate = target > displayedBands[i] and 26 or 7
        displayedBands[i] = displayedBands[i] + (target - displayedBands[i]) * (1 - math.exp(-math.min(dt, 0.25) * rate))
        local height = 2 + math.floor(26 * displayedBands[i])
        bar.Size, bar.Position = UDim2.fromOffset(5, height), UDim2.fromOffset((i - 1) * 7, 28 - height)
        bar.BackgroundColor3, bar.BackgroundTransparency = accent, 0.85 - 0.7 * displayedBands[i]
    end
    local character = Players.LocalPlayer and Players.LocalPlayer.Character
    local root = character and character:FindFirstChild("HumanoidRootPart")
    if morph.model then
        if not enforceMorphParts() then return end
        if morph.root ~= root or not root or not root.Parent then clearMorph()
        else
            if morph.poseDirty then
                morph.pose=buildMorphPose(morph.model,morph.pose.root,character)
                morph.poseDirty=false
            end
            if morph.pose then applyMorphPose(morph.pose,morphFrame(root)) end
            for object, saved in pairs(morph.hidden) do
                if object.Parent then pcall(function() object[saved.property] = 1 end) end
            end
        end
    end
    updateCharacterFx(character, root, accent)
    if skyPart and root and settings.sky and settings.lyrics and (skyLine ~= "" or previousSkyLine ~= "") then
        if not skyOrigin or skyRoot ~= root then
            skyRoot, skyOrigin = root, root.Position
            local look = root.CFrame.LookVector
            local flat = Vector3.new(look.X, 0, look.Z)
            skyDirection = flat.Magnitude > 0.01 and flat.Unit or Vector3.new(0, 0, -1)
        end
        skyPart.Position = skyOrigin + skyDirection * settings.skyDistance + Vector3.new(0, settings.skyHeight, 0)
        skyGui.Enabled = true
        local fade = settings.lyricMode~="Line" and math.min(settings.fade,0.12) or settings.fade
        local alpha = math.clamp((now - skyTransition) / fade, 0, 1)
        skyNew.TextTransparency = skyLine ~= "" and (1 - alpha) or 1
        skyOld.TextTransparency = previousSkyLine ~= "" and alpha or 1
        skyNew.TextStrokeTransparency = skyLine ~= "" and (1 - alpha * 0.7) or 1
        skyOld.TextStrokeTransparency = previousSkyLine ~= "" and (0.3 + alpha * 0.7) or 1
        skyNew.TextColor3, skyOld.TextColor3 = settings.skyColor, settings.skyColor
        skyNew.TextSize, skyOld.TextSize = settings.skySize, settings.skySize
        if alpha >= 1 then previousSkyLine = "" end
    else
        skyGui.Enabled = false
    end
end)

task.spawn(function()
    if type(requestFn) ~= "function" then offline = "Executor HTTP unavailable"; return end
    while alive do
        local success, response = pcall(requestFn, {Url = "http://127.0.0.1:8787/now-playing", Method = "GET"})
        if not alive then break end
        local code = success and type(response) == "table" and (response.StatusCode or response.status_code)
        if success and type(response) == "table" and (not code or code == 200) then
            local decoded, value = pcall(function() return HttpService:JSONDecode(response.Body or response.body) end)
            if decoded and type(value) == "table" then data, receivedAt = value, os.clock()
            else data, offline = nil, "Bridge response invalid" end
        else data, offline = nil, "Bridge offline" end
        task.wait(1)
    end
end)

-- Executors load remote cover bytes through a local asset, rather than an HTTP ImageLabel URL.
task.spawn(function()
    local loaded, tried, nextAttempt = "", "", 0
    local assetFn = getcustomasset or getsynasset
    if type(requestFn) ~= "function" or type(assetFn) ~= "function" or type(writefile) ~= "function" then return end
    local files = {}
    while alive do
        local path = data and data.album_art or ""
        if path ~= loaded then artworkImage.Visible = false end
        if path ~= "" and (path ~= tried or (path ~= loaded and os.clock() >= nextAttempt)) and path:match("^/album%-art%?key=%x+$") then
            tried, nextAttempt = path, os.clock() + 30
            local success, response = pcall(requestFn, {Url = "http://127.0.0.1:8787" .. path, Method = "GET"})
            if not alive then break end
            if success and type(response) == "table" and (response.StatusCode or response.status_code) == 200 then
                local bytes = response.Body or response.body
                local headers = response.Headers or response.headers or {}
                local mime = headers["Content-Type"] or headers["content-type"] or "image/jpeg"
                local name = "Vitality/Assets/gs-cover-" .. path:match("key=(%x+)") .. (mime:find("png", 1, true) and ".png" or ".jpg")
                local saved, asset = pcall(function()
                    assert(type(bytes) == "string" and #bytes <= 2000000)
                    writefile(name, bytes); return assetFn(name)
                end)
                if saved and data and data.album_art == path and alive then
                    artworkImage.Image, artworkImage.Visible, loaded = asset, true, path
                    table.insert(files, name)
                    if #files > 20 and type(delfile) == "function" then pcall(delfile, table.remove(files, 1)) end
                end
            end
        elseif path == "" then tried, loaded = "", "" end
        task.wait(1)
    end
end)

-- Refreshing this lease keeps audio transcription enabled only while the client is alive.
task.spawn(function()
    while alive do
        if liveDesired and liveRegistered and type(requestFn)=="function" then
            local success,response=pcall(requestFn,{Url="http://127.0.0.1:8787/live-lyrics?client="..liveOwner,Method="GET"})
            if not alive then break end
            if liveDesired then
                local code=success and type(response)=="table" and (response.StatusCode or response.status_code)
                local decoded,value=false,nil
                if success and type(response)=="table" and (not code or code==200) then
                    decoded,value=pcall(function() return HttpService:JSONDecode(response.Body or response.body) end)
                end
                if decoded and type(value)=="table" and value.enabled then
                    liveData,liveReceivedAt,liveTrackKey=value,os.clock(),currentTrackKey()
                    liveStatus=tostring(value.status or "listening")
                elseif decoded and type(value)=="table" and value.enabled==false then
                    liveData,liveRegistered,liveStatus=nil,false,"starting"
                    syncListener()
                else
                    liveData,liveStatus=nil,"live bridge unavailable"
                end
            end
        elseif liveDesired and not liveRegistered and os.clock()>=liveRetryAt then
            syncListener()
        end
        task.wait(0.5)
    end
end)
