-- The pinned library's title editor treats its title container as a TextLabel.
local function fixLibraryTitle(source)
    local broken = 'SettingsPage.Window.Items["ActualTitle"].Instance.Text = Value'
    local first, last = source:find(broken, 1, true)
    if first then
        local replacement = [[local container = SettingsPage.Window.Items["ActualTitle"].Instance
                        if container:IsA("TextLabel") then container.Text = Value else
                            local assigned = false
                            for _, child in ipairs(container:GetChildren()) do
                                if child:IsA("TextLabel") then
                                    child.Text = assigned and "" or Value
                                    assigned = true
                                end
                            end
                        end]]
        source = source:sub(1, first - 1) .. replacement .. source:sub(last + 1)
    end
    return source
end

local liveDesired,liveRegistered,liveSyncing,liveRetryAt=false,false,false,0
local liveData,liveReceivedAt,liveStatus,liveTrackKey=nil,0,"off",""
local function currentTrackKey()
    return data and data.token_ok and (tostring(data.song or "").."\0"..tostring(data.artist or "").."\0"..tostring(data.album or "")) or ""
end
local function syncListener()
    if liveSyncing or liveDesired==liveRegistered then return end
    if liveDesired and not bridgeAvailable() then liveRetryAt=os.clock()+5; return end
    if type(requestFn)~="function" then liveStatus="executor HTTP unavailable"; return end
    liveSyncing=true
    task.spawn(function()
        while liveDesired~=liveRegistered do
            local target=liveDesired
            local ok,response=pcall(function()
                return requestFn({Url="http://127.0.0.1:8787/live-lyrics",Method="POST",
                    Headers={["Content-Type"]="application/json"},Body=HttpService:JSONEncode({client=liveOwner,enabled=target})})
            end)
            local decoded,result=false,nil
            if ok and type(response)=="table" then decoded,result=pcall(function() return HttpService:JSONDecode(response.Body or response.body) end) end
            local code=ok and type(response)=="table" and (response.StatusCode or response.status_code)
            if not decoded or type(result)~="table" or not result.ok or (code and code~=200) then
                liveStatus=target and "live bridge unavailable" or "off"
                liveRetryAt=os.clock()+5
                break
            end
            liveRegistered=target
            liveStatus=target and "starting" or "off"
        end
        liveSyncing=false
    end)
end
local function setLiveListening(enabled)
    liveDesired=enabled and alive or false
    if not liveDesired then liveData=nil; liveTrackKey=""; liveStatus="off" end
    syncListener()
end

local skyOrigin, skyDirection, skyRoot = nil, nil, nil
local skyAnchorKey = ""
local function updateSkyAnchor(root, camera)
    local following = settings.skyAnchor ~= "Per line"
    local source
    if settings.skyAnchor == "Camera" then source = camera else source = root end
    if not source then return nil end
    local position = settings.skyAnchor == "Camera" and source.CFrame.Position or source.Position
    local needsAnchor = not skyOrigin or skyRoot ~= source
    if following and skyOrigin then
        needsAnchor = needsAnchor or settings.skyRadius == 0 or (position - skyOrigin).Magnitude > settings.skyRadius
    end
    if needsAnchor then
        skyRoot, skyOrigin = source, position
        local look = source.CFrame.LookVector
        local flat = Vector3.new(look.X, 0, look.Z)
        skyDirection = flat.Magnitude > 0.01 and flat.Unit or Vector3.new(0, 0, -1)
    end
    return skyOrigin + skyDirection * settings.skyDistance + Vector3.new(0, settings.skyHeight, 0)
end

local function parkEnvironment(object)
    if object ~= worldSky and object ~= worldAir and (object:IsA("Sky") or object:IsA("Atmosphere")) then
        savedEnvironment[object] = true
        object.Parent = nil
    end
end
local function restoreWorld()
    worldSky.Parent, worldAir.Parent = nil, nil
    for object in pairs(savedEnvironment) do
        pcall(function() if object.Parent == nil then object.Parent = Lighting end end)
    end
    savedEnvironment = {}
    if savedClock ~= nil and Lighting.ClockTime == imposedClock then Lighting.ClockTime = savedClock end
    savedClock, imposedClock = nil, nil
end
local function applyWorld()
    if not settings.world then restoreWorld(); return end
    if savedClock == nil then savedClock = Lighting.ClockTime end
    for _, object in ipairs(Lighting:GetChildren()) do parkEnvironment(object) end
    local preset = worldPresets[settings.worldPreset] or worldPresets["Violet night"]
    for index, face in ipairs({"Bk", "Dn", "Ft", "Lf", "Rt", "Up"}) do
        worldSky["Skybox" .. face] = "rbxassetid://" .. preset.textures[index]
    end
    worldSky.StarCount, worldSky.CelestialBodiesShown = settings.stars, true
    worldAir.Color, worldAir.Decay = preset.color, preset.decay
    worldAir.Density, worldAir.Haze, worldAir.Glare = settings.haze, 1.4, 0.15
    imposedClock = preset.time
    Lighting.ClockTime = imposedClock
    worldSky.Parent, worldAir.Parent = Lighting, Lighting
end
connect(Lighting.ChildAdded, function(object) if settings.world then parkEnvironment(object) end end)

-- Restore each camera's own value when the override is disabled or replaced.
local fovEnabled, fovValue, fovCamera, previousFov, appliedFov = false, 75, nil, nil, nil
local function restoreFov()
    if fovCamera and previousFov ~= nil then
        pcall(function() if fovCamera.FieldOfView == appliedFov then fovCamera.FieldOfView = previousFov end end)
    end
    fovCamera, previousFov, appliedFov = nil, nil, nil
end
local function applyFov()
    local camera = workspace.CurrentCamera
    if not fovEnabled or not camera then restoreFov(); return end
    if camera ~= fovCamera then
        restoreFov(); fovCamera, previousFov = camera, camera.FieldOfView
    end
    appliedFov = fovValue
    camera.FieldOfView = appliedFov
end

-- Apply third person after the game's camera and first-person visibility updates.
local thirdPerson, thirdDistance, thirdGeneration = false, 10, 0
local cameraSnapshot, zoomSnapshot = nil, nil
local hiddenViewParts, shownCharacterParts, shownBodyTransparency, hiddenViewEffects = {}, {}, {}, {}
local hiddenCandidates, shownCandidates, effectCandidates = {}, {}, {}
local thirdFighterModel, thirdFighterView, thirdFighterController, fighterCharacter = nil, nil, nil, nil
local lastViewScan = -math.huge
local thirdBinding = "LarpsenseThirdPerson_" .. liveOwner
local rayParams = RaycastParams.new()
rayParams.FilterType, rayParams.IgnoreWater = Enum.RaycastFilterType.Exclude, true
local bodyNames = {
    Head=true, Torso=true, UpperTorso=true, LowerTorso=true,
    ["Left Arm"]=true, ["Right Arm"]=true, ["Left Leg"]=true, ["Right Leg"]=true,
    LeftUpperArm=true, LeftLowerArm=true, LeftHand=true, RightUpperArm=true, RightLowerArm=true, RightHand=true,
    LeftUpperLeg=true, LeftLowerLeg=true, LeftFoot=true, RightUpperLeg=true, RightLowerLeg=true, RightFoot=true,
}
local function restoreLocalProperty(parts, property, imposed)
    for object, original in pairs(parts) do
        pcall(function() if object.Parent and object[property] == imposed then object[property] = original end end)
        parts[object] = nil
    end
end
local function restoreThirdCamera()
    if cameraSnapshot then
        local saved = cameraSnapshot
        pcall(function()
            if saved.appliedFrame and saved.camera.CFrame == saved.appliedFrame then saved.camera.CFrame = saved.inputFrame end
            if saved.camera.CameraType == Enum.CameraType.Custom then saved.camera.CameraType = saved.kind end
            if saved.imposedSubject and saved.camera.CameraSubject == saved.imposedSubject and (not saved.subject or saved.subject.Parent) then
                saved.camera.CameraSubject = saved.subject
            end
        end)
        cameraSnapshot = nil
    end
end
local function restoreThirdPerson()
    pcall(function() RunService:UnbindFromRenderStep(thirdBinding) end)
    restoreThirdCamera()
    if zoomSnapshot then
        local player, saved = Players.LocalPlayer, zoomSnapshot
        pcall(function()
            if player.CameraMode == Enum.CameraMode.Classic then player.CameraMode = saved.mode end
            if player.CameraMinZoomDistance == saved.applied and player.CameraMaxZoomDistance == saved.applied then
                player.CameraMaxZoomDistance = math.max(player.CameraMaxZoomDistance, saved.max)
                player.CameraMinZoomDistance = saved.min
                player.CameraMaxZoomDistance = saved.max
            end
        end)
        zoomSnapshot = nil
    end
    restoreLocalProperty(hiddenViewParts, "LocalTransparencyModifier", 1)
    restoreLocalProperty(shownCharacterParts, "LocalTransparencyModifier", 0)
    restoreLocalProperty(shownBodyTransparency, "Transparency", 0)
    for object, original in pairs(hiddenViewEffects) do
        pcall(function() if object.Parent and object.Enabled == false then object.Enabled = original end end)
        hiddenViewEffects[object] = nil
    end
    table.clear(hiddenCandidates); table.clear(shownCandidates); table.clear(effectCandidates)
    thirdFighterModel, thirdFighterView = nil, nil
end
local function liveInstance(value)
    return typeof(value) == "Instance" and value.Parent ~= nil
end
-- Resolve Rivals' local fighter outside the render callback; its visible model and
-- equipped viewmodel may live outside LocalPlayer.Character and CurrentCamera.
task.spawn(function()
    while alive do
        if thirdPerson then
            local generation = thirdGeneration
            local player = Players.LocalPlayer
            local character = player and player.Character
            if player and fighterCharacter ~= character then
                fighterCharacter, thirdFighterController, thirdFighterModel, thirdFighterView = player.Character, nil, nil, nil
            end
            pcall(function()
                if not thirdFighterController then
                    local scripts=player and player:FindFirstChild("PlayerScripts")
                    local controllers=scripts and scripts:FindFirstChild("Controllers")
                    local module=controllers and controllers:FindFirstChild("FighterController")
                    if module then
                        local controller=require(module)
                        if not alive or not thirdPerson or generation ~= thirdGeneration or Players.LocalPlayer ~= player or (player and player.Character ~= character) then return end
                        thirdFighterController=controller
                    end
                end
                local fighter=thirdFighterController and thirdFighterController.LocalFighter
                if not fighter or (fighter.Player and fighter.Player ~= player) then
                    if thirdFighterModel or thirdFighterView then lastViewScan=-math.huge end
                    thirdFighterModel,thirdFighterView=nil,nil
                    return
                end
                local oldModel, oldView = thirdFighterModel, thirdFighterView
                local entity=fighter.Entity
                local model=entity and (entity.Model or (entity.GetModel and entity:GetModel()))
                if not liveInstance(model) and fighter.Get then model=fighter:Get("Model") end
                if not liveInstance(model) then model=fighter.Model end
                if not alive or not thirdPerson or generation ~= thirdGeneration or Players.LocalPlayer ~= player or (player and player.Character ~= character) then return end
                thirdFighterModel=liveInstance(model) and model:IsA("Model") and model or nil
                local equipped=fighter.EquippedItem
                local view=equipped and (equipped.ViewModel or equipped.Viewmodel)
                if type(view)=="table" and not liveInstance(view) then view=view.Model or view.Instance end
                thirdFighterView=liveInstance(view) and view or nil
                if oldModel ~= thirdFighterModel or oldView ~= thirdFighterView then lastViewScan=-math.huge end
            end)
        end
        task.wait(0.25)
    end
end)
local function updateThirdPerson()
    if not thirdPerson then return end
    local player = Players.LocalPlayer
    if not player then return end
    if fighterCharacter ~= player.Character then
        fighterCharacter, thirdFighterModel, thirdFighterView = player.Character, nil, nil
        lastViewScan = -math.huge
    end
    if not zoomSnapshot then zoomSnapshot={mode=player.CameraMode,min=player.CameraMinZoomDistance,max=player.CameraMaxZoomDistance} end
    local saved = zoomSnapshot
    player.CameraMode = Enum.CameraMode.Classic
    player.CameraMaxZoomDistance = math.max(player.CameraMaxZoomDistance, thirdDistance)
    player.CameraMinZoomDistance = math.min(player.CameraMinZoomDistance, thirdDistance)
    player.CameraMinZoomDistance, player.CameraMaxZoomDistance = thirdDistance, thirdDistance
    saved.applied = thirdDistance
    local camera, character = workspace.CurrentCamera, player.Character
    local model = liveInstance(thirdFighterModel) and thirdFighterModel or character
    if cameraSnapshot and cameraSnapshot.camera ~= camera then restoreThirdCamera(); lastViewScan=-math.huge end
    if camera then
        if not cameraSnapshot then cameraSnapshot={camera=camera,kind=camera.CameraType,subject=camera.CameraSubject} end
        local subject = model and (model:FindFirstChildOfClass("Humanoid") or model:FindFirstChild("HumanoidRootPart",true) or model:FindFirstChild("Head",true))
        camera.CameraType = Enum.CameraType.Custom
        if subject then camera.CameraSubject = subject; cameraSnapshot.imposedSubject = subject end
    end
    local now = os.clock()
    if now - lastViewScan >= 0.1 then
        lastViewScan = now
        local seenHidden, seenShown, seenEffects = {}, {}, {}
        local function isExplicitView(object)
            return liveInstance(thirdFighterView) and (object==thirdFighterView or object:IsDescendantOf(thirdFighterView))
        end
        local function isBody(object)
            if isExplicitView(object) then return false end
            return (character and (object==character or object:IsDescendantOf(character))) or
                (model and (object==model or object:IsDescendantOf(model)))
        end
        local function visit(root, ownedCharacter)
            if not root then return end
            local objects = root:GetDescendants()
            table.insert(objects, root)
            for _, object in ipairs(objects) do
                if (ownedCharacter and not isExplicitView(object)) or (not ownedCharacter and not isBody(object)) then
                    if object:IsA("BasePart") or object:IsA("Decal") then
                        if ownedCharacter then
                            if not object:IsA("BasePart") or object.Transparency < 1 or bodyNames[object.Name] then seenShown[object]=true end
                        else seenHidden[object]=true end
                    elseif not ownedCharacter and (object:IsA("ParticleEmitter") or object:IsA("Beam") or object:IsA("Trail") or object:IsA("BillboardGui") or object:IsA("SurfaceGui") or object:IsA("Highlight")) then
                        seenEffects[object]=true
                    end
                end
            end
        end
        visit(character,true)
        if model and model ~= character then visit(model,true) end
        visit(camera,false)
        if liveInstance(thirdFighterView) then visit(thirdFighterView,false) end
        local folder=workspace:FindFirstChild("ViewModels")
        if folder then
            for _, child in ipairs(folder:GetChildren()) do
                local name=child.Name:lower()
                if name:find(player.Name:lower(),1,true) or name=="viewmodel" or name=="arms" then visit(child,false) end
            end
        end
        local single=workspace:FindFirstChild("ViewModel")
        if single and single ~= character and single ~= model then visit(single,false) end
        for object, original in pairs(hiddenViewParts) do
            if not seenHidden[object] then pcall(function() if object.Parent and object.LocalTransparencyModifier==1 then object.LocalTransparencyModifier=original end end); hiddenViewParts[object]=nil end
        end
        for object, original in pairs(shownCharacterParts) do
            if not seenShown[object] then pcall(function() if object.Parent and object.LocalTransparencyModifier==0 then object.LocalTransparencyModifier=original end end); shownCharacterParts[object]=nil end
        end
        for object, original in pairs(shownBodyTransparency) do
            if not seenShown[object] then pcall(function() if object.Parent and object.Transparency==0 then object.Transparency=original end end); shownBodyTransparency[object]=nil end
        end
        for object, original in pairs(hiddenViewEffects) do
            if not seenEffects[object] then pcall(function() if object.Parent and object.Enabled==false then object.Enabled=original end end); hiddenViewEffects[object]=nil end
        end
        hiddenCandidates,shownCandidates,effectCandidates=seenHidden,seenShown,seenEffects
    end
    -- The game can overwrite transparency every frame; enforce cached parts after it runs.
    for object in pairs(shownCandidates) do
        pcall(function()
            if object.Parent then
                if object.LocalTransparencyModifier ~= 0 then
                    if shownCharacterParts[object]==nil then shownCharacterParts[object]=object.LocalTransparencyModifier end
                    object.LocalTransparencyModifier=0
                end
                if object:IsA("BasePart") and bodyNames[object.Name] and object.Transparency>=1 then
                    if shownBodyTransparency[object]==nil then shownBodyTransparency[object]=object.Transparency end
                    object.Transparency=0
                end
            end
        end)
    end
    for object in pairs(hiddenCandidates) do
        pcall(function()
            if object.Parent and object.LocalTransparencyModifier ~= 1 then
                if hiddenViewParts[object]==nil then hiddenViewParts[object]=object.LocalTransparencyModifier end
                object.LocalTransparencyModifier=1
            end
        end)
    end
    for object in pairs(effectCandidates) do
        pcall(function()
            if object.Parent and object.Enabled then
                if hiddenViewEffects[object]==nil then hiddenViewEffects[object]=true end
                object.Enabled=false
            end
        end)
    end
    if camera and model then
        local focusPart = model:FindFirstChild("Head",true) or model:FindFirstChild("HumanoidRootPart",true)
        if focusPart and focusPart:IsA("BasePart") then
            local focus = focusPart.Position
            if focusPart.Name == "HumanoidRootPart" then focus=focus+Vector3.new(0,1.5,0) end
            local inputFrame=camera.CFrame
            if not cameraSnapshot.appliedFrame or inputFrame ~= cameraSnapshot.appliedFrame then cameraSnapshot.inputFrame=inputFrame end
            local look=inputFrame.LookVector
            local distance=thirdDistance
            local ignored={camera}
            if character then table.insert(ignored,character) end
            if model ~= character then table.insert(ignored,model) end
            if liveInstance(thirdFighterView) then table.insert(ignored,thirdFighterView) end
            local folder=workspace:FindFirstChild("ViewModels")
            local single=workspace:FindFirstChild("ViewModel")
            if folder then table.insert(ignored,folder) end
            if single then table.insert(ignored,single) end
            rayParams.FilterDescendantsInstances=ignored
            local ok,hit=pcall(function() return workspace:Raycast(focus, -look*distance, rayParams) end)
            if ok and hit then distance=math.max(0.5,math.min(distance,(hit.Position-focus).Magnitude-0.35)) end
            camera.CFrame=CFrame.lookAt(focus-look*distance,focus)
            cameraSnapshot.appliedFrame=camera.CFrame
        end
    end
end

local function stopManagedAnimation(owner)
    if owner and animationOwner ~= owner then return end
    animationGeneration += 1
    if animationTrack then pcall(function() animationTrack:Stop(0.15); animationTrack:Destroy() end); animationTrack=nil end
    if animationObject then animationObject:Destroy(); animationObject=nil end
    animationOwner=nil
end
local function resolveEmoteAnimation(id)
    if resolvedAnimations[id] then return resolvedAnimations[id] end
    local asset = "rbxassetid://" .. tostring(id)
    -- Catalog emote assets can contain a separate Animation instance.
    local ok, objects = pcall(function() return game:GetObjects(asset) end)
    local found
    if ok and type(objects)=="table" then
        for _, object in ipairs(objects) do
            pcall(function()
                local animation = object:IsA("Animation") and object or object:FindFirstChildWhichIsA("Animation", true)
                local assetNumber = animation and type(animation.AnimationId)=="string" and animation.AnimationId:match("%d+")
                if not found and assetNumber and tonumber(assetNumber) and tonumber(assetNumber)>0 then found=animation.AnimationId end
            end)
            pcall(function() object:Destroy() end)
        end
    end
    if found then resolvedAnimations[id]=found end
    return found or asset
end
local function animationFailed(owner, message, detail)
    stopManagedAnimation(owner)
    if owner == "spin" then
        spinningActive=false
        if spinToggleControl then spinToggleControl:Set(false) end
    end
    EmoteStatus:SetText(message)
    warn("Larpsense: " .. message .. (detail and (" " .. tostring(detail)) or ""))
end
local function playManagedAnimation(id, owner, looped, displayName)
    stopManagedAnimation()
    animationOwner = owner
    local generation = animationGeneration
    local character = Players.LocalPlayer and Players.LocalPlayer.Character
    EmoteStatus:SetText("Loading: " .. displayName)
    task.spawn(function()
        local animationId = resolveEmoteAnimation(id)
        if not alive or generation ~= animationGeneration or character ~= Players.LocalPlayer.Character then return end
        local humanoid = character and character:FindFirstChildOfClass("Humanoid")
        local animator = humanoid and humanoid:FindFirstChildOfClass("Animator")
        if not animator then
            animationFailed(owner, "No character animator available.")
            return
        end
        local animation=Instance.new("Animation")
        animation.AnimationId=animationId
        local ok, track = pcall(function() return animator:LoadAnimation(animation) end)
        if not alive or generation ~= animationGeneration or character ~= Players.LocalPlayer.Character then
            animation:Destroy(); if ok and track then pcall(function() track:Destroy() end) end; return
        end
        if not ok or not track then
            animation:Destroy()
            animationFailed(owner, "Could not load: " .. displayName, track)
            return
        end
        animationObject, animationTrack = animation, track
        local played, playError = pcall(function()
            track.Priority=Enum.AnimationPriority.Action4
            track.Looped=looped
            track:Play(0.15)
        end)
        if not played then
            animationFailed(owner, "Could not play: " .. displayName, playError)
            return
        end
        EmoteStatus:SetText("Playing: " .. displayName)
        track.Stopped:Connect(function()
            if alive and generation == animationGeneration and animationTrack == track then
                stopManagedAnimation(owner)
                if owner == "spin" then
                    spinningActive=false
                    if spinToggleControl then spinToggleControl:Set(false) end
                end
                EmoteStatus:SetText(displayName .. " stopped.")
            end
        end)
        task.delay(10, function()
            if alive and generation == animationGeneration and animationTrack == track and track.Length == 0 then
                EmoteStatus:SetText("Animation unavailable: " .. displayName)
                warn("Larpsense: " .. displayName .. " did not load. The game may restrict this animation.")
            end
        end)
    end)
end

local function findGunModel()
    local cam = workspace.CurrentCamera
    if not cam then return nil end
    local lp = LocalPlayer
    local best, bestScore = nil, 0

    local function score(m)
        if not m or not m:IsA("Model") then return 0 end
        if lp and (m == lp.Character or m:IsDescendantOf(lp.Character)) then return 0 end
        local wpn, arms, parts = 0, 0, 0
        for _, d in ipairs(m:GetDescendants()) do
            if d:IsA("BasePart") and d.Transparency < 1 then
                parts += 1
                local n = d.Name
                if nameHit(ARM_NAME, n) then arms += 1
                elseif nameHit(WPN_NAME, n) then wpn += 1 end
            end
        end
        if parts == 0 then return 0 end
        -- pure-ish gun model scores high; mixed still ok if any weapon parts
        local s = wpn * 5 - arms * 2
        if wpn == 0 and arms == 0 and parts >= 2 then
            -- unnamed mesh gun — weak candidate
            s = parts * 0.5
        end
        local nm = m.Name:lower()
        if lp and nm:find(lp.Name:lower(), 1, true) then s += 2 end
        if nm:find("view", 1, true) or nm:find("gun", 1, true) or nm:find("weapon", 1, true) then s += 3 end
        return s
    end

    local function scan(root, depth)
        if not root or depth > 3 then return end
        for _, child in ipairs(root:GetChildren()) do
            if child:IsA("Model") then
                local sc = score(child)
                if sc > bestScore then bestScore, best = sc, child end
                scan(child, depth + 1)
            elseif child:IsA("Folder") then
                scan(child, depth + 1)
            end
        end
    end

    scan(cam, 0)
    scan(workspace:FindFirstChild("ViewModels"), 0)
    scan(workspace:FindFirstChild("ViewModel"), 0)
    scan(workspace:FindFirstChild("Camera"), 0)

    if bestScore >= 1 then return best end
    return nil
end

local function paintGun(model, fill, fillT)
    if not model then return end
    local seen = {}
    local painted = 0
    for _, d in ipairs(model:GetDescendants()) do
        if d:IsA("BasePart") and d.Transparency < 1 then
            local n = d.Name
            if nameHit(ARM_NAME, n) then continue end
            local ok = nameHit(WPN_NAME, n)
            -- if model is mostly weapon, also paint generic mesh parts
            if not ok then
                local ln = n:lower()
                if ln == "meshpart" or ln == "part" or ln == "handle" or ln:find("mesh") then
                    ok = true
                end
            end
            if not ok then continue end
            seen[d] = true
            painted += 1
            if not _gcPainted[d] then
                _gcPainted[d] = { Color = d.Color, Material = d.Material, Transparency = d.Transparency }
            end
            pcall(function()
                d.Color = fill
                if S.gc.Style == "Glass" then
                    d.Material = Enum.Material.Glass
                elseif S.gc.Style == "ForceField" then
                    d.Material = Enum.Material.ForceField
                else
                    d.Material = Enum.Material.Neon
                end
                d.Transparency = math.clamp(fillT, 0, 0.85)
            end)
        end
    end
    for p, data in pairs(_gcPainted) do
        if not seen[p] then
            pcall(function()
                if p.Parent and data then
                    p.Color = data.Color
                    p.Material = data.Material
                    p.Transparency = data.Transparency
                end
            end)
            _gcPainted[p] = nil
        end
    end
    -- highlight whole model as backup when few named parts
    local hl = getGunHL()
    if painted < 2 then
        hl.Adornee = model
        hl.FillColor = fill

local function applyVM()
    if thirdPerson then
        for part in pairs(_vmOrig) do restoreVMPart(part); _vmOrig[part] = nil end
        return
    end
    local cam = workspace.CurrentCamera
    if not cam then return end
    local roots = { cam }
    local vmF = workspace:FindFirstChild("ViewModels")
    if vmF then table.insert(roots, vmF) end
    local seen = {}
    for _, root in ipairs(roots) do
        for _, d in ipairs(root:GetDescendants()) do
            if d:IsA("BasePart") then
                local n = d.Name:lower()
                local isVM = n:find("arm") or n:find("hand") or n:find("finger") or n:find("sleeve")
                    or n:find("glove") or n:find("weapon") or n:find("gun") or n:find("view")
                    or (d.Parent and d.Parent:IsA("Model") and d.Parent.Parent == cam)
                if not isVM and root == cam then isVM = true end
                if isVM then
                    seen[d] = true
                    if not _vmOrig[d] then
                        _vmOrig[d] = { Color = d.Color, Material = d.Material, Transparency = d.Transparency, Reflectance = d.Reflectance }
                    end
                    if S.vm.Enabled then
                        local t  = (math.sin(tick() * (S.vm.Speed or 3)) + 1) * 0.5
                        local sty = S.vm.Style
                        local col
                        if sty == "Rainbow" then
                            col = Color3.fromHSV((tick() * 0.2) % 1, 1, 1)
                        elseif sty == "Blackout" then
                            col = Color3.new(0, 0, 0)
                        else
                            col = S.vm.Color1:Lerp(S.vm.Color2, t)
                        end
                        if sty == "Hot" or sty == "Neon" or sty == "Rainbow" then
                            d.Material     = Enum.Material.Neon
                            d.Color        = col
                            d.Transparency = S.vm.Transparency
                            d.Reflectance  = 0.1
                        elseif sty == "ForceField" then
                            d.Material     = Enum.Material.ForceField
                            d.Color        = col
                            d.Transparency = S.vm.Transparency
                        elseif sty == "Glass" then
                            d.Material     = Enum.Material.Glass
                            d.Color        = col
                            d.Transparency = math.max(S.vm.Transparency, 0.25)
                            d.Reflectance  = 0.5
                        elseif sty == "Blackout" then
                            d.Material     = Enum.Material.SmoothPlastic
                            d.Color        = col
                            d.Transparency = S.vm.Transparency
                            d.Reflectance  = 0
                        else -- Smooth
                            d.Material     = Enum.Material.SmoothPlastic
                            d.Color        = col
                            d.Transparency = S.vm.Transparency
                        end
                    else
                        restoreVMPart(d)
                    end
                end
            end
        end
    end
    for p in pairs(_vmOrig) do
        if not seen[p] or not p.Parent then
            restoreVMPart(p)
            _vmOrig[p] = nil
        end
    end
end
fireConnect(RunService.RenderStepped, function() pcall(applyVM) end)

task.spawn(function()
    local ok, err = pcall(function()
        local rep = game:GetService("ReplicatedStorage")
        local plrs = game:GetService("Players")
        local runs = game:GetService("RunService")
        local cs = game:GetService("CollectionService")
        local lp = plrs.LocalPlayer
        local cam = workspace.CurrentCamera

        local U = require(rep.Modules.Utility)
        local EL = require(rep.Modules.EnumLibrary)
        local GU = require(rep.Modules.GameplayUtility)
        if not fireAlive then return end

        pcall(function()
            local ok2, gun = pcall(require, lp.PlayerScripts.Modules.ItemTypes.Gun)
            if ok2 and gun and gun.IsFullyAiming then
                local oldAim = gun.IsFullyAiming
                local wrapper = function(...)
                    if fireAlive and S.saim.Enabled then return true end
                    return oldAim(...)
                end
                gun.IsFullyAiming = wrapper
                registerFireCleanup(function() if gun.IsFullyAiming == wrapper then gun.IsFullyAiming = oldAim end end)
            end
        end)

        local FOVFill = newFireDrawing("Circle")
        FOVFill.Filled = true
        FOVFill.NumSides = 64
        FOVFill.ZIndex = 1
        FOVFill.Visible = false

        local FOV = newFireDrawing("Circle")
        FOV.Filled = false
        FOV.NumSides = 64
        FOV.Thickness = 1
        FOV.ZIndex = 3
        FOV.Visible = false

        local Snap = newFireDrawing("Line")
        Snap.Thickness = 1
        Snap.ZIndex = 3
        Snap.Visible = false

        local function hideAllFov()
            FOVFill.Visible = false
            FOV.Visible = false
            Snap.Visible = false
        end

        local function target()
            if not S.saim.Enabled then return nil end
            cam = workspace.CurrentCamera
            if not cam then return nil end

            local center = cam.ViewportSize / 2
            local fov = (getgenv().Config and getgenv().Config.FOVRadius) or S.saim.FOVRadius or 300
            local hitName = (getgenv().Config and getgenv().Config.HitPart) or S.saim.HitPart or "Head"
            local myRoot = lp.Character and lp.Character:FindFirstChild("HumanoidRootPart")
            local origin = myRoot and myRoot.Position or cam.CFrame.Position
            local bestPart, bestDist = nil, fov

            local function teamOk(plr)
                if not S.saim.TeamCheck then return true end
                local pT = plr:GetAttribute("TeamID")
                local lT = lp:GetAttribute("TeamID")
                if pT ~= nil and lT ~= nil then return pT ~= lT end
                if plr.Team and lp.Team then return plr.Team ~= lp.Team end
                return true
            end

            local function resolveModel(plr)
                -- rivals body is often an Entity model, not Character
                for _, e in ipairs(cs:GetTagged("Entity")) do
                    if typeof(e) == "Instance" and e:IsA("Model") then
                        local n = e.Name
                        if n == plr.Name or n:find(plr.Name, 1, true) then
                            return e
                        end
                        local attr = e:GetAttribute("UserId") or e:GetAttribute("PlayerUserId")
                        if attr and tonumber(attr) == plr.UserId then
                            return e
                        end
                    end
                end
                if plr.Character and plr.Character.Parent then
                    return plr.Character
                end
                return nil
            end

            local function alive(model, plr)
                local hum = model:FindFirstChildOfClass("Humanoid")
                if hum then return hum.Health > 0 end
                if plr.Character then
                    local h2 = plr.Character:FindFirstChildOfClass("Humanoid")
                    if h2 then return h2.Health > 0 end
                end
                return model:FindFirstChild("Head") ~= nil
            end

            local function losOk(part)
                if not S.saim.WallCheck then return true end
                local params = RaycastParams.new()
                params.FilterType = Enum.RaycastFilterType.Exclude
                params.FilterDescendantsInstances = { lp.Character, cam }
                params.IgnoreWater = true
                local dir = part.Position - origin
                if dir.Magnitude < 1 then return true end
                local hit = workspace:Raycast(origin, dir, params)
                if not hit then return true end
                return hit.Instance and hit.Instance:IsDescendantOf(part.Parent)
            end

            local function getHitPart(model)
                local part = model:FindFirstChild(hitName, true)
                if part and part:IsA("BasePart") then return part end
                for _, n in ipairs({ "Head", "UpperTorso", "HumanoidRootPart", "Torso" }) do
                    local p = model:FindFirstChild(n, true)
                    if p and p:IsA("BasePart") then return p end
                end
                return nil
            end

            local fov = (getgenv().Config and getgenv().Config.FOVRadius) or S.saim.FOVRadius or 300
            bestPart, bestDist = nil, math.huge

            for _, plr in ipairs(plrs:GetPlayers()) do
                if plr == lp then continue end
                if not teamOk(plr) then continue end
                local model = resolveModel(plr)
                if not model then continue end
                if not alive(model, plr) then continue end
                local part = getHitPart(model)
                if not part then continue end
                if part.Transparency >= 0.99 then continue end
                local worldDist = (part.Position - origin).Magnitude
                if worldDist > (S.saim.MaxDist or 1500) then continue end
                if not losOk(part) then continue end

                local sp = cam:WorldToViewportPoint(part.Position)
                if sp.Z <= 0 then continue end
                local screenDist = (Vector2.new(sp.X, sp.Y) - center).Magnitude
                if screenDist > fov then continue end

                if worldDist < bestDist then
                    bestDist = worldDist
                    bestPart = part
                end
            end

            return bestPart
        end

        local function CFD(origin, part)
            local aimPos = part.Position
            local look = CFrame.lookAt(origin, aimPos)
            local cf = part.CFrame
            local d = {}
            d[utf8.char(1)] = {
                [utf8.char(0)] = U:EncodeCFrame(look),
                [utf8.char(1)] = U:EncodeCFrame(S.saim.Manipulate and look or cf),
                [utf8.char(2)] = part,
                [utf8.char(3)] = U:EncodeCFrame(cf:ToObjectSpace(CFrame.new(aimPos))),
            }
            return d
        end

        local lastTP = nil

        fireConnect(runs.RenderStepped, function()
            cam = workspace.CurrentCamera
            if not cam then hideAllFov() lastTP = nil return end
            if not (S.saim.Enabled and S.saim.ShowFOV) then
                hideAllFov()
                lastTP = nil
                return
            end

            local t = target()
            lastTP = t and t.Position or nil

            local radius = (getgenv().Config and getgenv().Config.FOVRadius) or S.saim.FOVRadius or 300
            if S.saim.Anim then
                radius = radius + math.sin(tick() * (S.saim.PulseSpeed or 4)) * 6
            end

            local center = cam.ViewportSize / 2
            local pos = center
            if S.saim.FollowTarget and lastTP then
                local sp, on = cam:WorldToViewportPoint(lastTP)
                if on and sp.Z > 0 then pos = Vector2.new(sp.X, sp.Y) end
            end

            local hue = (tick() * 0.35) % 1
            local col = S.saim.FOVColor or Color3.new(1,1,1)
            if S.saim.Rainbow then col = Color3.fromHSV(hue, 1, 1) end
            local fillCol = S.saim.FOVFillColor or Color3.fromRGB(255, 80, 120)
            if S.saim.Rainbow then fillCol = Color3.fromHSV((hue + 0.15) % 1, 0.9, 1) end
            local fillA = math.clamp(S.saim.FOVFillAlpha or 0.65, 0, 1)
            local sides = S.saim.Sides or 64
            local thick = S.saim.Thickness or 1

            FOVFill.Position = pos
            FOVFill.Radius = radius
            FOVFill.Color = fillCol
            FOVFill.Transparency = fillA
            FOVFill.NumSides = sides
            FOVFill.Visible = S.saim.FOVFill == true

            FOV.Position = pos
            FOV.Radius = radius
            FOV.Color = col
            FOV.Thickness = thick
            FOV.NumSides = sides
            FOV.Visible = true

            if S.saim.Snapline and lastTP then
                local sp, on = cam:WorldToViewportPoint(lastTP)
                if on and sp.Z > 0 then
                    Snap.From = center
                    Snap.To = Vector2.new(sp.X, sp.Y)
                    Snap.Color = col
                    Snap.Thickness = thick
                    Snap.Visible = true
                else
                    Snap.Visible = false
                end
            else
                Snap.Visible = false
            end
        end)

        local visual = GU.GetEntitiesFromRaycast
        local visualWrapper = function(self, envID, params, origin, dir, maxDist, ...)
            if fireAlive and S.saim.Enabled then
                local t = target()
                if t then
                    local dist = (t.Position - origin).Magnitude
                    dir = (t.Position - origin).Unit
                    if dist > maxDist then maxDist = dist + 5 end
                end
            end
            return visual(self, envID, params, origin, dir, maxDist, ...)
        end
        GU.GetEntitiesFromRaycast = visualWrapper
        registerFireCleanup(function() if GU.GetEntitiesFromRaycast == visualWrapper then GU.GetEntitiesFromRaycast = visual end end)

        local UseItem = rep.Remotes.Replication.Fighter.UseItem
        local startEnum = EL:ToEnum("StartShooting")
        requestShootHook = function()
            if not fireAlive or tracerHookAvailable then return end
            local oldFS
            local installed, original = pcall(function()
                return hookfunction(UseItem.FireServer, newcclosure(function(self, ...)
                    -- FireServer is shared by other remotes: forward every argument unchanged.
                    if not fireAlive or self ~= UseItem or select("#", ...) < 2 or select(2, ...) ~= startEnum then
                        return oldFS(self, ...)
                    end
                    if S.sfx.Shoot or S.sfx.Hit or S.sfx.Kill then markShot() end
                    local camdata = select("#", ...) >= 3 and select(3, ...) or nil
                    if S.saim.Enabled and not rageFiring and select("#", ...) >= 3 then
                        local chance = S.saim.HitChance or 100
                        if chance >= 100 or math.random(1, 100) <= chance then
                            local root = lp.Character and lp.Character:FindFirstChild("HumanoidRootPart")
                            local targetOK, part = pcall(target)
                            if root and targetOK and part then
                                local encoded, replacement = pcall(CFD, root.Position, part)
                                if encoded and replacement ~= nil then
                                    camdata = replacement
                                    local args = table.pack(...)
                                    args[3] = replacement
                                    if S.tracerShot then pcall(S.tracerShot, camdata) end
                                    return oldFS(self, table.unpack(args, 1, args.n))
                                end
                            end
                        end
                    end
                    if S.tracerShot then pcall(S.tracerShot, camdata) end
                    return oldFS(self, ...)
                end))
            end)
            if installed and type(original) == "function" then
                oldFS, tracerHookAvailable = original, true
            else
                warn("Larpsense: shooting hook unavailable; standard Tool tracers will still work.")
            end
        end
        if S.saim.Enabled or (S.tracer2 and S.tracer2.Enabled) or S.sfx.Shoot or S.sfx.Hit or S.sfx.Kill then requestShootHook() end
    end)
end)

    local function patchItem(item)
        if not item then return end
        pcall(function()
            local info = rawget(item,"Info") or rawget(item,"_data")
            if type(info) == "table" then
                if S.gun.NoRecoil then
                    for _, k in ipairs({"ShootRecoil","Recoil","CameraKick","RecoilX","RecoilY"}) do

                    end

                        if info[k] ~= nil then info[k] = 0 end
                    end
                end
                if S.gun.FastReload then
                    for _, k in ipairs({"ReloadTime","EmptyReload","ReloadSpeed"}) do
                        if info[k] ~= nil then info[k] = 0.05 end
                    end
                end
                if S.gun.FireRate then
                    local m = S.gun.FireRateMul or 0.15
                    for _, k in ipairs({"ShootCooldown","FireCooldown","AttackCooldown"}) do
                        if type(info[k]) == "number" then
                            info["_base_"..k] = info["_base_"..k] or info[k]
                            info[k] = info["_base_"..k] * m
                        end
                    end
                end
            end
        end)
    end

    while fireAlive do
        if S.gun.NoRecoil or S.gun.NoSpread or S.gun.FastReload or S.gun.FireRate then
            pcall(function()
                local fc = require(LocalPlayer.PlayerScripts.Controllers.FighterController)
                local lf = fc and fc.LocalFighter
                if lf and lf.EquippedItem then patchItem(lf.EquippedItem) end
            end)
            if getgc then
                for _, obj in ipairs(getgc(true)) do
                    if type(obj) == "table" then
                        pcall(function()
                            if S.gun.NoSpread then
                                if rawget(obj,"SpreadMagnitude") then rawset(obj,"SpreadMagnitude",0) end
                                if rawget(obj,"BaseSpread")      then rawset(obj,"BaseSpread",0)      end
                                if rawget(obj,"BulletSpread")    then rawset(obj,"BulletSpread",0)    end
                            end
                            if S.gun.NoRecoil then
                                if rawget(obj,"ShootRecoil") then rawset(obj,"ShootRecoil",0) end
                                if rawget(obj,"CameraKick")  then rawset(obj,"CameraKick",0)  end
                            end
                        end)
                    end
                end
            end
        end
        task.wait(0.2)
    end
end)

task.spawn(function()
    -- Rivals may load its modules and remotes after the UI; retry until ready.
    local repS, plrs, runS, ws, util, enum, FC, useItem
    local attempts, warned = 0, false
    while fireAlive do
        local ok, result = pcall(function()
            local function ref(object) return type(cloneref)=="function" and cloneref(object) or object end
            local rep=ref(game:GetService("ReplicatedStorage"))
            local players=ref(game:GetService("Players"))
            local runs=ref(game:GetService("RunService"))
            local world=ref(game:GetService("Workspace"))
            local localPlayer=players.LocalPlayer
            local modules=rep:FindFirstChild("Modules")
            local utility=modules and modules:FindFirstChild("Utility")
            local enums=modules and modules:FindFirstChild("EnumLibrary")
            local scripts=localPlayer and localPlayer:FindFirstChild("PlayerScripts")
            local controllers=scripts and scripts:FindFirstChild("Controllers")
            local controller=controllers and controllers:FindFirstChild("FighterController")
            local remotes=rep:FindFirstChild("Remotes")
            local replication=remotes and remotes:FindFirstChild("Replication")
            local fighter=replication and replication:FindFirstChild("Fighter")
            local remote=fighter and fighter:FindFirstChild("UseItem")
            assert(utility and enums and controller and remote, "Rivals modules or weapon remote not ready")
            local u,e,fc=require(utility),require(enums),require(controller)
            assert(u and e and fc, "Rivals modules unavailable")
            return {rep,players,runs,world,u,e,fc,remote}
        end)
        if not fireAlive then return end
        if ok then
            repS,plrs,runS,ws,util,enum,FC,useItem=table.unpack(result)
            break
        end
        attempts += 1
        if S.rage.Enabled then setRageStatus("Waiting for Rivals") end
        if S.rage.Enabled and attempts >= 6 and not warned then
            warned = true
            warn("Larpsense ragebot: waiting for Rivals modules and weapon remote: " .. tostring(result))
        end
        task.wait(1)
    end
    if not fireAlive then return end
    setRageStatus(S.rage.Enabled and "Looking for a target" or "Ready")
    local ok2, err2 = pcall(function()
        local lplr = plrs.LocalPlayer
        local slots = { Primary=1, Secondary=2, Melee=3 }
        local function slot() return slots[getgenv().Config.WeaponSlot] or 3 end
        task.spawn(function()
            local lf = FC.LocalFighter
            while fireAlive and not lf do task.wait(0.1); lf=FC.LocalFighter end
            if fireAlive and lf and S.rage.Enabled then pcall(function() lf:EquipItem(slot()) end) end
        end)
        task.spawn(function()
            while fireAlive do task.wait(1)
                if fireAlive and getgenv().Config.Enabled then
                    local lf=FC.LocalFighter; if lf then pcall(function() lf:EquipItem(slot()) end) end
                end
            end
        end)
        local lastFire, lastFireWarning = 0, 0
        local fighters, taggedModels, scanGeneration = {}, {}, 0
        fireConnect(plrs.PlayerRemoving, function(player) fighters[player]=nil; taggedModels[player]=nil end)
        local function validModel(model)
            return typeof(model)=="Instance" and model:IsA("Model") and model.Parent ~= nil
        end
        local function readFighter(fo)
            local entity=fo and fo.Entity
            local model, aliveState, spectating, deflect = nil,nil,false,false
            if entity then
                pcall(function() if entity.IsAlive then aliveState=entity:IsAlive() end end)
                pcall(function() model=entity.Model or (entity.GetModel and entity:GetModel()) end)
            end
            if not validModel(model) then pcall(function() if fo.Get then model=fo:Get("Model") end end) end
            if not validModel(model) then pcall(function() model=fo.Model end) end
            if not validModel(model) then model=nil end
            pcall(function() if fo.Get then spectating=fo:Get("IsSpectating") == true end end)
            pcall(function()
                local eq=fo.EquippedItem
                local vm=eq and (eq.ViewModel or eq.Viewmodel)
                local name=vm and tostring(vm.Name) or ""
                if eq and eq.Get then name=name..tostring(eq:Get("Name") or eq:Get("ItemName") or "") end
                if name:lower():find("katana") or name:lower():find("sword") then
                    deflect=(eq._attack_cooldown and eq._attack_cooldown>tick()) or
                        (eq.Get and eq:Get("IsDeflecting")) or (eq._last_attack and tick()-eq._last_attack<0.35) or false
                end
            end)
            return {model=model,alive=aliveState,spectating=spectating,deflecting=deflect}
        end
        task.spawn(function()
            while fireAlive do
                if S.rage.Enabled then
                    local generation=scanGeneration
                    local nextFighters,nextTagged={},{}
                    pcall(function()
                        for _, fo in pairs(FC.Objects or {}) do
                            pcall(function()
                                local player=fo.Player
                                if player and player.Parent==plrs then
                                    local character=player.Character
                                    local info=readFighter(fo); info.character=character
                                    nextFighters[player]=info
                                end
                            end)
                        end
                        if FC.LocalFighter then
                            local character=lplr.Character
                            local info=readFighter(FC.LocalFighter); info.character=character
                            nextFighters[lplr]=info
                        end
                    end)
                    pcall(function()
                        local candidates=CollectionService:GetTagged("Entity")
                        for _, model in ipairs(candidates) do
                            if validModel(model) then
                                local userId=model:GetAttribute("UserId") or model:GetAttribute("PlayerUserId")
                                local playerName=model:GetAttribute("PlayerName") or model.Name
                                for _, player in ipairs(plrs:GetPlayers()) do
                                    if (userId and tonumber(userId)==player.UserId) or playerName==player.Name then nextTagged[player]=model; break end
                                end
                            end
                        end
                    end)
                    if fireAlive and S.rage.Enabled and generation==scanGeneration then fighters,taggedModels=nextFighters,nextTagged end
                end
                task.wait(0.25)
            end
        end)
        local function isEnemy(player)
            if not player or player == lplr then return false end
            if S.rage.TeamCheck and sameTeam(player) then return false end
            return true
        end
        local orbitTarget, orbitAngle, pendingProne = nil, nil, nil
        local restoreBinding = "LarpsenseOrbitRestore_" .. liveOwner
        local function restoreProne()
            local state = pendingProne
            pendingProne = nil
            if not state then return end
            pcall(function() runS:UnbindFromRenderStep(restoreBinding) end)
            pcall(function()
                if state.root.Parent and state.root.CFrame == state.applied then state.root.CFrame = state.normal end
            end)
            pcall(function()
                if state.humanoid and state.humanoid.Parent and state.humanoid.PlatformStand == true then
                    state.humanoid.PlatformStand = state.platformStand
                end
            end)
        end
        stopRageOrbit = function(invalidate)
            restoreProne()
            orbitTarget, orbitAngle = nil, nil
            if invalidate then
                scanGeneration += 1
                table.clear(fighters); table.clear(taggedModels)
            end
        end
        local function modelParts(model)
            if not validModel(model) then return nil end
            local root=model:FindFirstChild("HumanoidRootPart",true) or model:FindFirstChild("RootPart",true) or model.PrimaryPart
            local head=model:FindFirstChild("Head",true) or model:FindFirstChild("UpperTorso",true) or model:FindFirstChild("Torso",true) or root
            if not (root and root:IsA("BasePart") and head and head:IsA("BasePart")) then return nil end
            return root,head,model:FindFirstChildWhichIsA("Humanoid",true)
        end
        local function targetParts(player, myRoot)
            if not isEnemy(player) or player.Parent ~= plrs then return nil end
            local fighter=fighters[player]
            if fighter and fighter.character ~= player.Character then fighter=nil end
            if fighter and (fighter.alive==false or fighter.spectating or (S.rage.KatanaCheck and fighter.deflecting)) then return nil end
            local model=fighter and fighter.model
            if not validModel(model) then model=taggedModels[player] end
            if not validModel(model) then model=player.Character end
            local root,head,humanoid=modelParts(model)
            if not root and model ~= player.Character then root,head,humanoid=modelParts(player.Character) end
            if not root then return nil end
            if humanoid and humanoid.Health <= 0 then return nil end
            local characterHumanoid=player.Character and player.Character:FindFirstChildWhichIsA("Humanoid")
            if not humanoid and fighter and fighter.alive ~= true and characterHumanoid and characterHumanoid.Health <= 0 then return nil end
            local distance=(myRoot.Position-root.Position).Magnitude
            if distance > (S.rage.MaxDist or 500) then return nil end
            return root,head,distance
        end
        local function getTarget(myRoot)
            if orbitTarget then
                local root,head=targetParts(orbitTarget,myRoot)
                if root then return orbitTarget,root,head end
            end
            local nearest,nearestRoot,nearestHead,best=nil,nil,nil,math.huge
            for _,player in ipairs(plrs:GetPlayers()) do
                local root,head,distance=targetParts(player,myRoot)
                if root and distance<best then nearest,nearestRoot,nearestHead,best=player,root,head,distance end
            end
            if nearest ~= orbitTarget then orbitAngle=nil end
            orbitTarget=nearest
            return nearest,nearestRoot,nearestHead
        end
        fireConnect(runS.Heartbeat, function(dt)
            restoreProne()
            if not fireAlive or not S.rage.Enabled then stopRageOrbit(); return end
            if not getgenv().Config then getgenv().Config = {} end
            getgenv().Config.Enabled = true
            local character=lplr.Character
            local myR,_,myH=modelParts(character)
            local localFighter=fighters[lplr]
            if localFighter and localFighter.character ~= character then localFighter=nil end
            if localFighter and (localFighter.alive==false or localFighter.spectating) then stopRageOrbit(); setRageStatus("Waiting for character"); return end
            if not myR and localFighter then myR,_,myH=modelParts(localFighter.model) end
            if not myR or (myH and myH.Health <= 0) then stopRageOrbit(); setRageStatus("Waiting for character"); return end
            local tp,tr,th=getTarget(myR)
            if not tp then stopRageOrbit(); setRageStatus("Looking for a target"); return end
            if orbitAngle == nil then
                local delta=myR.Position-tr.Position
                orbitAngle=math.atan2(delta.Z,delta.X)
            end
            orbitAngle=(orbitAngle + math.rad(S.rage.OrbitSpeed)*math.clamp(dt or 0,0,0.1)) % (math.pi*2)
            local radius=S.rage.OrbitRadius
            local orbitPosition=tr.Position + Vector3.new(math.cos(orbitAngle)*radius,0,math.sin(orbitAngle)*radius)
            local dCF=CFrame.lookAt(orbitPosition,th.Position)
            myR.CFrame=dCF
            if S.rage.LayDown and myH then
                local tilted=dCF*CFrame.Angles(math.rad(-90),0,0)
                pendingProne={root=myR,humanoid=myH,normal=dCF,applied=tilted,platformStand=myH.PlatformStand}
                myR.CFrame=tilted
                myH.PlatformStand=true
                runS:BindToRenderStep(restoreBinding,101,restoreProne)
            end
            if not (S.rage.Enabled or (getgenv().Config and getgenv().Config.Enabled)) or not tp or not th or not tr then return end
            if not FC.LocalFighter then setRageStatus("Waiting for fighter"); return end
            local item=FC.LocalFighter.EquippedItem
            if not item then setRageStatus("Waiting for weapon"); return end
            local idOK, objectId=pcall(function() return item:Get("ObjectID") end)
            if not idOK or objectId==nil then setRageStatus("Weapon unavailable"); return end
            if tick()-lastFire < getgenv().Config.FireRate then return end
            lastFire=tick()
            local fired, fireError=pcall(function()
                local originP=dCF.Position
                local tPos=th.Position
                local aCF=CFrame.lookAt(originP,tPos); local tCF=th.CFrame
                local rOff=Vector3.new((math.random()-0.5)*0.1,(math.random()-0.5)*0.1,(math.random()-0.5)*0.1)
                local aimed=tPos+rOff
                local osHO=th.CFrame:ToObjectSpace(CFrame.new(aimed))
                local cd2={}
                cd2[utf8.char(1)]={
                    [utf8.char(0)]=util:EncodeCFrame(aCF),
                    [utf8.char(1)]=util:EncodeCFrame(tCF),
                    [utf8.char(2)]=th,
                    [utf8.char(3)]=util:EncodeCFrame(osHO),
                }
                rageFiring=true
                useItem:FireServer(objectId,enum:ToEnum("StartShooting"),cd2,nil)
            end)
            rageFiring=false
            if fired then
                setRageStatus("Orbiting " .. tp.Name)
            else
                setRageStatus("Could not fire weapon")
                if tick()-lastFireWarning>=5 then lastFireWarning=tick(); warn("Larpsense ragebot: weapon call failed: " .. tostring(fireError)) end
            end
        end)
    end)
    if not ok2 and fireAlive then
        setRageStatus("Setup failed; see console")
        warn("Larpsense ragebot setup failed: " .. tostring(err2))
    end
end)

local function applyMove()
    local char = LocalPlayer.Character
    if not char then return end
    local hum = char:FindFirstChildOfClass("Humanoid")
    local hrp = char:FindFirstChild("HumanoidRootPart")
    if not hum or not hrp then return end

    if S.move.Noclip then
        for _, p in ipairs(char:GetDescendants()) do
            if p:IsA("BasePart") then
                if savedCollision[p] == nil then savedCollision[p] = p.CanCollide end
                p.CanCollide = false
            end
        end
    end

    if S.move.Jump then
        hum.JumpPower = S.move.JumpVal
        pcall(function() hum.JumpHeight = S.move.JumpVal / 5 end)
    end

    local want = nil
    if S.move.Sprint and _sprint then
        want = S.move.SprintVal
    elseif S.move.Speed then
        want = S.move.SpeedVal
    end
    if want and not S.move.Fly then
        if math.abs(hum.WalkSpeed - want) > 0.1 then
            hum.WalkSpeed = want
        end
    end

    if S.move.Fly then
        local cam = workspace.CurrentCamera
        local dir = Vector3.zero
        if cam then
            local look = cam.CFrame.LookVector
            local right = cam.CFrame.RightVector
            if UserInputService:IsKeyDown(Enum.KeyCode.W) then dir += look end
            if UserInputService:IsKeyDown(Enum.KeyCode.S) then dir -= look end
            if UserInputService:IsKeyDown(Enum.KeyCode.A) then dir -= right end
            if UserInputService:IsKeyDown(Enum.KeyCode.D) then dir += right end
            if UserInputService:IsKeyDown(Enum.KeyCode.Space) then dir += Vector3.yAxis end
            if UserInputService:IsKeyDown(Enum.KeyCode.LeftShift) then dir -= Vector3.yAxis end
        end
        if dir.Magnitude > 0 then
            hrp.AssemblyLinearVelocity = dir.Unit * S.move.FlySpeed
        else
            hrp.AssemblyLinearVelocity = Vector3.zero
        end
        hum.PlatformStand = false
    end
end

end)

task.spawn(function()
    pcall(function()
        local mcM = LocalPlayer.PlayerScripts.Controllers:WaitForChild("MechanicsController", 10)
        if not mcM then return end
        local MC = require(mcM)
        local mt = getmetatable(MC)
        if not mt or type(mt.__index) ~= "table" then return end
        local idx = mt.__index
        local oldGet = rawget(idx, "_GetWalkSpeed") or rawget(idx, "GetWalkSpeed")
        if type(oldGet) ~= "function" then
            for k, v in pairs(idx) do
                if type(v) == "function" and tostring(k):lower():find("walkspeed") then
                    oldGet = v
                    break
                end
            end
        end
        if not fireAlive or type(oldGet) ~= "function" then return end
        local function hooked(self, ...)
            local speed = oldGet(self, ...)
            if not fireAlive or type(speed) ~= "number" then return speed end
            if S.move.Speed then
                speed = S.move.SpeedVal
            end
            if S.move.Sprint and _sprint then
                speed = S.move.SprintVal
            end
            if S.move.SlideMulti then
                local sliding = false
                pcall(function()
                    if self.IsSliding then sliding = self:IsSliding() end
                    if self.Get and self:Get("IsSliding") then sliding = true end
                end)
                if not sliding then
                    local char = LocalPlayer.Character
                    local hum = char and char:FindFirstChildOfClass("Humanoid")
                    if hum and (hum:GetAttribute("IsSliding") or char:GetAttribute("IsSliding")) then
                        sliding = true
                    elseif hum and hum.FloorMaterial ~= Enum.Material.Air then
                        if UserInputService:IsKeyDown(Enum.KeyCode.C) or UserInputService:IsKeyDown(Enum.KeyCode.LeftControl) then
                            local hrp = char:FindFirstChild("HumanoidRootPart")
                            if hrp and hrp.AssemblyLinearVelocity.Magnitude > 18 then sliding = true end
                        end
                    end
                end
                if sliding then speed = speed * (S.move.SlideMultiVal or 10) end
            end
            return speed
        end
        local oldPrivate, oldPublic = rawget(idx, "_GetWalkSpeed"), rawget(idx, "GetWalkSpeed")
        rawset(idx, "_GetWalkSpeed", hooked)
        if oldPublic then rawset(idx, "GetWalkSpeed", hooked) end
        registerFireCleanup(function()
            if rawget(idx, "_GetWalkSpeed") == hooked then rawset(idx, "_GetWalkSpeed", oldPrivate) end
            if rawget(idx, "GetWalkSpeed") == hooked then rawset(idx, "GetWalkSpeed", oldPublic) end
        end)
    end)
end)

local _entityByPlayer = {}
local _playerByModel = {}
local _fcCache = nil
local _fcAt = 0

local function getFighterController()
    if _fcCache and tick() - _fcAt < 2 then return _fcCache end
    pcall(function()
        local mod = LocalPlayer.PlayerScripts.Controllers:FindFirstChild("FighterController")
        if mod then
            _fcCache = require(mod)
            _fcAt = tick()
        end
    end)
    return _fcCache
end

local function refreshEntityMap()
    table.clear(_entityByPlayer)
    table.clear(_playerByModel)
    local FC = getFighterController()
    if FC and FC.Objects then
        for _, fo in pairs(FC.Objects) do
            pcall(function()
                local plr = fo.Player
                local ent = fo.Entity
                local model = ent and (ent.Model or (ent.GetModel and ent:GetModel()))
                if not model and fo.Get and fo:Get("Model") then model = fo:Get("Model") end
                if not model and typeof(fo.Model) == "Instance" then model = fo.Model end
                if plr and model then
                    _entityByPlayer[plr] = model
                    _playerByModel[model] = plr
                end
            end)
        end
    end
    pcall(function()
        for _, model in ipairs(CollectionService:GetTagged("Entity")) do
            if model:IsA("Model") then
                local plr = _playerByModel[model]
                if not plr then
                    for _, p in ipairs(Players:GetPlayers()) do
                        if p ~= LocalPlayer and p.Character == model then
                            plr = p
                            break
                        end
                        if p ~= LocalPlayer and (model.Name == p.Name or model:GetAttribute("PlayerName") == p.Name) then
                            plr = p
                            break
                        end
                    end
                end
                if plr then
                    _entityByPlayer[plr] = model
                    _playerByModel[model] = plr
                end
            end
        end
    end)
end

local function resolveEspModel(plr)
    local m = _entityByPlayer[plr]
    if m and m.Parent then return m end
    local ch = plr.Character
    if ch and ch.Parent then return ch end
    return nil
end

-- Geometry is shared by the external editor and actual player overlays.
local function espSide(position)
    local side = tostring(position or ""):match("^%a+")
    if side == "Top" or side == "Bottom" or side == "Left" or side == "Right" then return side end
    return "Top"
end
local function healthGeometry(x,y,w,h,position)
    local side = espSide(position)
    if side == "Top" then return x,y-6,w,3,true end
    if side == "Bottom" then return x,y+h+3,w,3,true end
    if side == "Right" then return x+w+3,y,3,h,false end
    return x-6,y,3,h,false
end
local function labelLayout(x,y,w,h,textSize,healthSide,healthOn)
    local health = healthOn and espSide(healthSide) or nil
    local cursor={Top=y-2-(health=="Top" and 5 or 0),Bottom=y+h+2+(health=="Bottom" and 5 or 0),Left=y+h*0.5-4,Right=y+h*0.5-4}
    return function(pos, object, width, height)
        width=math.max(width,math.ceil(#object.Text*textSize*0.55))
        height=math.max(height,textSize+2)
        local side,px,py=espSide(pos),nil,nil
        local cur=cursor[side]
        if side=="Top" then py=cur-height; px=x+w*0.5-width*0.5; cursor[side]=py-1
        elseif side=="Bottom" then py=cur; px=x+w*0.5-width*0.5; cursor[side]=cur+height+1
        elseif side=="Left" then py=cur; px=x-width-4-(health=="Left" and 5 or 0); cursor[side]=cur+height+1
        else py=cur; px=x+w+4+(health=="Right" and 5 or 0); cursor[side]=cur+height+1 end
        object.Position,object.Size=UDim2.fromOffset(math.floor(px),math.floor(py)),UDim2.fromOffset(width,height)
        object.TextSize=textSize
    end
end

-- Restore local visibility when chams stop or a player leaves range.
local function restoreChamVisibility(s)
    for part, original in pairs(s.chamVisibility or {}) do
        pcall(function()
            if part.Parent and part.LocalTransparencyModifier == 0 then part.LocalTransparencyModifier = original end
        end)
    end
    s.chamVisibility, s.chamVisibilityAt = nil, nil
end
local function revealChamParts(char, s)
    local now = os.clock()
    if s.chamVisibilityAt and now - s.chamVisibilityAt < 0.5 then return end
    s.chamVisibilityAt = now
    s.chamVisibility = s.chamVisibility or {}
    for _, part in ipairs(char:GetDescendants()) do
        if part:IsA("BasePart") and part.LocalTransparencyModifier > 0.05 then
            if s.chamVisibility[part] == nil then s.chamVisibility[part] = part.LocalTransparencyModifier end
            part.LocalTransparencyModifier = 0
        end
    end
end
local function hideEsp(s)
    s.box.Visible=false; s.hpBg.Visible=false; s.fill.Visible=false
    s.name.Visible=false; s.user.Visible=false; s.dist.Visible=false
    s.hpNum.Visible=false; s.status.Visible=false; s.vis.Visible=false
    s.wep.Visible=false; s.ping.Visible=false; s.tracer.Visible=false; s.headDot.Visible=false
    for _, line in ipairs(s.skel) do line.Visible=false end
end
local function hideAll(s)
    hideEsp(s)
    restoreChamVisibility(s)
    if s.hl then s.hl.Enabled=false end
    if s.aura then s.aura.Enabled=false end
    if s.wires then for _, box in pairs(s.wires) do pcall(function() box:Destroy() end) end; s.wires={} end
end
local function destroyEspSet(s)
    hideAll(s)
    for _, key in ipairs({"box", "hpBg", "name", "user", "dist", "hpNum", "status", "vis", "wep", "ping", "tracer", "headDot", "hl", "aura"}) do
        if s[key] then s[key]:Destroy() end
    end
    for _, line in ipairs(s.skel) do line:Destroy() end
end

local _espRayParams = RaycastParams.new()
_espRayParams.FilterType = Enum.RaycastFilterType.Exclude
_espRayParams.IgnoreWater = true

local function isVisible(origin, char)
    local tgt = char:FindFirstChild("Head") or char:FindFirstChild("HumanoidRootPart")
    if not tgt then return false end
    local dir = tgt.Position - origin
    if dir.Magnitude < 1 then return true end
    _espRayParams.FilterDescendantsInstances = { LocalPlayer.Character, char, workspace.CurrentCamera }
    local hit = workspace:Raycast(origin, dir, _espRayParams)
    return not hit or (hit.Instance and hit.Instance:IsDescendantOf(char))
end

local function drawLine(f,a,b,color,thick)
    local dx,dy=b.X-a.X,b.Y-a.Y; local len=math.sqrt(dx*dx+dy*dy)
    if len<1 then f.Visible=false; return end
    f.Size=UDim2.fromOffset(len,thick or 1); f.Position=UDim2.fromOffset((a.X+b.X)/2,(a.Y+b.Y)/2)
    f.Rotation=math.deg(math.atan2(dy,dx)); f.BackgroundColor3=color; f.Visible=true
end

local function getState(hum)
    if not hum then return "Standing" end
    local st=hum:GetState()
    if st==Enum.HumanoidStateType.Running then return "Walking"  end
    if st==Enum.HumanoidStateType.Jumping  then return "Jumping"  end
    if st==Enum.HumanoidStateType.Freefall then return "Falling"  end
    if st==Enum.HumanoidStateType.Climbing then return "Climbing" end
    if st==Enum.HumanoidStateType.Swimming then return "Swimming" end
    return "Standing"
end

local function updatePlayerChams(plr, char, hrp, s)
    if S.Chams then revealChamParts(char, s) else restoreChamVisibility(s) end
    local fillCol = S.TeamColor and plr.TeamColor and plr.TeamColor.Color or S.ChamFill
    local outCol  = S.TeamColor and plr.TeamColor and plr.TeamColor.Color or S.ChamOut
    local fillT, outT = S.ChamFillT, S.ChamOutT
    local sty = S.ChamStyle
    if sty=="Hot" then
        local t=(math.sin(tick()*4)+1)*0.5
        fillCol=Color3.fromRGB(255,90,30):Lerp(Color3.fromRGB(255,40,200),t)
        outCol=fillCol
    elseif sty=="Rainbow" then
        fillCol=Color3.fromHSV((tick()*0.25)%1,1,1); outCol=fillCol; fillT=0.15; outT=0
    elseif sty=="Neon" then
        fillT=0.05; outT=0
    elseif sty=="Outline" then
        fillT=1; outT=0
    elseif sty=="Glow" then
        fillT=0.4; outT=0.15
    elseif sty=="Pulse" then
        fillT=math.clamp(0.15+math.sin(tick()*4)*0.25, 0, 1); outT=0
    elseif sty=="Wire" then
        fillT=1; outT=0
    end

    if S.Chams then
        if not s.hl or not s.hl.Parent then
            local hl=Instance.new("Highlight")
            hl.Name="FP_Cham_"..char.Name
            hl.Adornee=char
            hl.DepthMode=Enum.HighlightDepthMode.AlwaysOnTop
            hl.Parent=CoreGui
            s.hl=hl
        end
        pcall(function()
            s.hl.Adornee = char
            s.hl.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
            s.hl.FillColor = fillCol
            s.hl.OutlineColor = outCol
            s.hl.FillTransparency = fillT
            s.hl.OutlineTransparency = outT
            s.hl.Enabled = true
        end)
    elseif s.hl then
        s.hl.Enabled = false
    end

    if S.Chams and sty=="Wire" then
        s.wires=s.wires or {}; local want={}
        for _, p in ipairs(char:GetChildren()) do
            if p:IsA("BasePart") and p.Name~="HumanoidRootPart" and p.Transparency<1 then
                want[p]=true
                local box=s.wires[p]
                if not box or not box.Parent then
                    box=Instance.new("SelectionBox")
                    box.LineThickness=0.025
                    box.SurfaceTransparency=1
                    box.Parent=CoreGui
                    s.wires[p]=box
                end
                box.Adornee=p
                box.Color3=outCol
                box.Visible=true
            end
        end
        for p, box in pairs(s.wires) do
            if not want[p] then pcall(function() box:Destroy() end); s.wires[p]=nil end
        end
    elseif s.wires then
        for p, box in pairs(s.wires) do pcall(function() box:Destroy() end) end
        s.wires={}
    end

    if S.Aura then
        if not s.aura then
            s.aura=Instance.new("ParticleEmitter"); s.aura.Texture="rbxassetid://243098098"
            s.aura.Rate=20; s.aura.Lifetime=NumberRange.new(0.5,1); s.aura.Speed=NumberRange.new(1,3)
            s.aura.SpreadAngle=Vector2.new(180,180); s.aura.Size=NumberSequence.new(S.AuraSize)
            s.aura.Transparency=NumberSequence.new(0.5); s.aura.Parent=hrp
        end
        s.aura.Color=ColorSequence.new(S.ChamFill); s.aura.Enabled=true
    elseif s.aura then s.aura.Enabled=false end
end

local function muzzlePosition(camera, character)
    local tool = character and character:FindFirstChildOfClass("Tool")
    local viewmodels = workspace:FindFirstChild("ViewModels")
    for _, container in ipairs({tool or camera, viewmodels or camera}) do
        for _, name in ipairs({"Muzzle", "MuzzleAttachment", "FirePoint"}) do
            local muzzle = container:FindFirstChild(name, true)
            if muzzle then
                local pos = muzzle:IsA("Attachment") and muzzle.WorldPosition or (muzzle:IsA("BasePart") and muzzle.Position)
                if pos and (pos - camera.CFrame.Position).Magnitude < 25 then return pos end
            end
        end
    end
    return camera.CFrame.Position + camera.CFrame.LookVector * 1.5
end

S.tracerShot = function(camdata)
    if not S.tracer2.Enabled then return end

        if typeof(part) == "Instance" and part:IsA("BasePart") then destination = part.Position end
    end

local function clock(ms)
    local seconds = math.floor(math.max(0, tonumber(ms) or 0) / 1000)
    return string.format("%d:%02d", math.floor(seconds / 60), seconds % 60)
end
local function lyricAt(lines, seconds)
    local low, high, found = 1, #lines, 0
    while low <= high do
        local mid = math.floor((low + high) / 2)
        if (tonumber(lines[mid].time) or 0) <= seconds then found, low = mid, mid + 1 else high = mid - 1 end
    end
    return found
end
-- Tagged words are absolute seconds; otherwise interpolate evenly within the current line.
local function joinWords(words, last)
    local result=""
    for i=1,last do
        local word=words[i]
        if word and word~="" then
            result=result .. ((result~="" and not word:match("^[%.,!?;:]")) and " " or "") .. word
        end
    end
    return result
end
local function lyricWord(lines,index,seconds,duration,buildLine)
    local line=lines[index]
    if not line then return "","unavailable" end
    local start=tonumber(line.time)
    local following=lines[index+1] and tonumber(lines[index+1].time) or duration
    if not start or not following or following<=start then return "","unavailable" end
    local tagged=line.words
    if type(tagged)=="table" and #tagged>0 then
        local valid,previous,selected,values=true,start,0,{}
        for _,entry in ipairs(tagged) do
            local t=type(entry)=="table" and tonumber(entry.time) or nil
            local value=type(entry)=="table" and entry.text or nil
            if not t or t<previous or t>=following or type(value)~="string" or value=="" then valid=false; break end
            previous=t
            values[#values+1]=value
            if t<=seconds then selected=#values end
        end
        if valid then return buildLine and joinWords(values,selected) or (values[selected] or ""),"tagged" end
    end
    local words={}
    for word in tostring(line.text or ""):gmatch("%S+") do words[#words+1]=word end
    if #words==0 then return "","unavailable" end
    local fraction=math.clamp((seconds-start)/(following-start),0,1)
    local selected=math.min(#words,math.floor(fraction*#words)+1)
    return buildLine and joinWords(words,selected) or words[selected],"estimated"
end

                end

                if settings.lyricMode=="Line" then shownLyric=actualLine
                elseif settings.lyricMode=="Word" then shownLyric=words[selected] or " "
                else shownLyric=selected>0 and joinWords(words,selected) or " " end

        end

        task.delay(4, function() pcall(function() loadingGui:Destroy() end) end)
    end
else
    setLoading(featureLoadFailed and "Some features failed. Check the console." or "Ready", 1)
    task.delay(featureLoadFailed and 2.5 or 0.5, function() pcall(function() loadingGui:Destroy() end) end)
end
